```typescript
// src/memdir/memoryScan.ts
type MemoryHeader = {
  filename: string          # 相对路径
  filePath: string          # 绝对路径
  mtimeMs: number           # 修改时间
  description: string | null # 前置描述
  type: MemoryType | undefined # 类型标签
}

const MAX_MEMORY_FILES = 200
const FRONTMATTER_MAX_LINES = 30

async function scanMemoryFiles(
  memoryDir: string,
  signal: AbortSignal
): Promise<MemoryHeader[]> {
  // 1. 读取目录
  const entries = await readdir(memoryDir, { recursive: true })

  // 2. 过滤 .md 文件 (排除 MEMORY.md)
  const mdFiles = entries.filter(
    f => f.endsWith('.md') &&
         basename(f) !== 'MEMORY.md' &&
         (f.split(sep).length - 1) < MAX_DEPTH  // 限制深度 3 层
  )

  // 3. 解析前置
  const headers = await Promise.allSettled(
    mdFiles.map(async relativePath => {
      const filePath = join(memoryDir, relativePath)
      const { content, mtimeMs } = await readFileInRange(
        filePath, 0, FRONTMATTER_MAX_LINES
      )
      const { frontmatter } = parseFrontmatter(content)
      return {
        filename: relativePath,
        filePath,
        mtimeMs,
        description: frontmatter.description || null,
        type: parseMemoryType(frontmatter.type),
      }
    })
  )

  // 4. 按修改时间排序 (最新优先)
  return headers
    .filter(r => r.status === 'fulfilled')
    .map(r => r.value)
    .sort((a, b) => b.mtimeMs - a.mtimeMs)
    .slice(0, MAX_MEMORY_FILES)
}

// 格式化为清单
function formatMemoryManifest(memories: MemoryHeader[]): string {
  return memories.map(m => {
    const tag = m.type ? `[${m.type}] ` : ''
    const ts = new Date(m.mtimeMs).toISOString()
    return m.description
      ? `- ${tag}${m.filename} (${ts}): ${m.description}`
      : `- ${tag}${m.filename} (${ts})`
  }).join('\n')
}
```

```typescript
type LSPServerManager = {
  initialize(): Promise<void>
  shutdown(): Promise<void>
  getServerForFile(filePath: string): LSPServerInstance | undefined
  ensureServerStarted(filePath: string): Promise<LSPServerInstance | undefined>
  sendRequest<T>(filePath: string, method: string, params: unknown): Promise<T | undefined>
  getAllServers(): Map<string, LSPServerInstance>
  openFile(filePath: string, content: string): Promise<void>
  changeFile(filePath: string, content: string): Promise<void>
  saveFile(filePath: string): Promise<void>
  closeFile(filePath: string): Promise<void>
  isFileOpen(filePath: string): boolean
}

function createLSPServerManager(): LSPServerManager {
  const servers: Map<string, LSPServerInstance> = new Map()
  const extensionMap: Map<string, string[]> = new Map()  # ext → server names
  const openedFiles: Map<string, string> = new Map()     # URI → server name

  async function initialize(): Promise<void> {
    // 1. 加载配置
    const { servers: serverConfigs } = await getAllLspServers()

    // 2. 构建扩展名映射
    for (const [serverName, config] of Object.entries(serverConfigs)) {
      const fileExtensions = Object.keys(config.extensionToLanguage)
      for (const ext of fileExtensions) {
        const normalized = ext.toLowerCase()
        if (!extensionMap.has(normalized)) {
          extensionMap.set(normalized, [])
        }
        extensionMap.get(normalized)?.push(serverName)
      }

      // 3. 创建实例
      const instance = createLSPServerInstance(serverName, config)
      servers.set(serverName, instance)
    }
  }

  async function ensureServerStarted(filePath: string): Promise<LSPServerInstance | undefined> {
    const ext = path.extname(filePath).toLowerCase()
    const serverNames = extensionMap.get(ext)
    if (!serverNames?.length) return undefined

    const serverName = serverNames[0]
    const instance = servers.get(serverName)
    if (!instance) return undefined

    if (!instance.isRunning()) {
      await instance.start()
    }

    return instance
  }

  async function sendRequest<T>(
    filePath: string,
    method: string,
    params: unknown
  ): Promise<T | undefined> {
    const instance = await ensureServerStarted(filePath)
    if (!instance) return undefined

    const uri = pathToFileURL(filePath).href
    return instance.sendRequest<T>(method, params)
  }
}
```

```typescript
// src/utils/tokenAnalytics.ts
type TokenUsageEntry = {
  timestamp: number
  inputTokens: number
  outputTokens: number
  cacheReadTokens: number
  cacheCreationTokens: number
  model: string
}

type TokenAnalytics = {
  totalRequests: number
  totalInputTokens: number
  totalOutputTokens: number
  totalCacheRead: number
  totalCacheCreation: number
  averageInputPerRequest: number
  averageOutputPerRequest: number
  cacheHitRate: number         # %
  mostUsedModel: string
  requestsLastHour: number
  requestsLastDay: number
}

class TokenUsageTracker {
  private history: TokenUsageEntry[] = []
  private readonly maxEntries: number = 1000

  record(usage: APIUsage): void {
    this.history.push({
      timestamp: Date.now(),
      inputTokens: usage.input_tokens,
      outputTokens: usage.output_tokens,
      cacheReadTokens: usage.cache_read_input_tokens ?? 0,
      cacheCreationTokens: usage.cache_creation_input_tokens ?? 0,
      model: usage.model,
    })

    // 限制历史大小
    if (this.history.length > this.maxEntries) {
      this.history = this.history.slice(-this.maxEntries)
    }
  }

  getAnalytics(): TokenAnalytics {
    // 聚合统计...
    const totalRequests = this.history.length
    let totalInput = 0, totalOutput = 0, totalCacheRead = 0

    for (const entry of this.history) {
      totalInput += entry.inputTokens
      totalOutput += entry.outputTokens
      totalCacheRead += entry.cacheReadTokens
    }

    const cacheHitRate = (totalCacheRead / (totalInput + totalOutput + totalCacheRead)) * 100

    return {
      totalRequests,
      totalInputTokens: totalInput,
      totalOutputTokens: totalOutput,
      totalCacheRead,
      averageInputPerRequest: Math.round(totalInput / totalRequests),
      averageOutputPerRequest: Math.round(totalOutput / totalRequests),
      cacheHitRate: Math.round(cacheHitRate),
      mostUsedModel: this.findMostUsedModel(),
      requestsLastHour: this.countSince(Date.now() - 60 * 60 * 1000),
      requestsLastDay: this.countSince(Date.now() - 24 * 60 * 60 * 1000),
    }
  }

  estimateCost(): { input: number; output: number; cache: number } {
    const analytics = this.getAnalytics()
    // 近似定价 (Claude 4 pricing)
    const inputCost = analytics.totalInputTokens * 0.00015  # $/token
    const outputCost = analytics.totalOutputTokens * 0.0006
    const cacheCost = analytics.totalCacheRead * 0.000075

    return { input: inputCost, output: outputCost, cache: cacheCost }
  }
}
```

```typescript
  // 1. 环境变量覆盖

async function tryGenerateSuggestion(
  abortController: AbortController,
  messages: Message[],
  getAppState: () => AppState,
  cacheSafeParams: CacheSafeParams
): Promise<{ suggestion: string; promptId: PromptVariant } | null> {
  // 1. 检查消息数量 (至少 2 个 assistant turns)
  const assistantTurnCount = count(messages, m => m.type === 'assistant')
  if (assistantTurnCount < 2) return null

  // 2. 检查 suppress 原因

  if (suppressReason) return null

  // 3. 生成建议

    abortController,
    promptId,
    cacheSafeParams
  )

  // 4. 过滤检查
  if (shouldFilterSuggestion(suggestion)) return null

  return { suggestion, promptId }
}
```

```typescript
    f => f.endsWith('.md') && basename(f) !== 'MEMORY.md'
  )

  const headers = await Promise.all(
    mdFiles.map(async relativePath => {
      const filePath = join(memoryDir, relativePath)
      const content = await readFile(filePath, 'utf8')
      const { frontmatter } = parseFrontmatter(content)
      const stats = await stat(filePath)

      return {
        filename: relativePath,
        filePath,
        mtimeMs: stats.mtimeMs,
        description: frontmatter.description || null,
        type: parseMemoryType(frontmatter.type),
      }
    })
  )

  return headers
    .sort((a, b) => b.mtimeMs - a.mtimeMs)
    .slice(0, MAX_MEMORY_FILES)
}

export function formatMemoryManifest(memories: MemoryHeader[]): string {
  return memories.map(m => {
    const tag = m.type ? `[${m.type}] ` : ''
    return m.description
      ? `- ${tag}${m.filename}: ${m.description}`
      : `- ${tag}${m.filename}`
  }).join('\n')
}
```

**OpenClaude 参考**: `src/services/lsp/LSPServerManager.ts`

```typescript
// src/lsp/manager.ts
export interface LSPServerConfig {
  command: string
  args?: string[]
  env?: Record<string, string>
  extensionToLanguage: Record<string, string>
}

export interface LSPServerInstance {
  start(): Promise<void>
  stop(): Promise<void>
  isRunning(): boolean
  sendRequest<T>(method: string, params: unknown): Promise<T>
}

export interface LSPServerManager {
  initialize(configs: Record<string, LSPServerConfig>): Promise<void>
  shutdown(): Promise<void>
  getServerForFile(filePath: string): LSPServerInstance | undefined
  sendRequest<T>(filePath: string, method: string, params: unknown): Promise<T | undefined>
}

export function createLSPServerManager(): LSPServerManager {
  const servers: Map<string, LSPServerInstance> = new Map()
  const extensionMap: Map<string, string[]> = new Map()

  async function initialize(configs: Record<string, LSPServerConfig>): Promise<void> {
    for (const [name, config] of Object.entries(configs)) {
      // 构建扩展名映射
      for (const ext of Object.keys(config.extensionToLanguage)) {
        if (!extensionMap.has(ext)) extensionMap.set(ext, [])
        extensionMap.get(ext)?.push(name)
      }

      // 创建实例
      const instance = createLSPServerInstance(name, config)
      servers.set(name, instance)
    }
  }

  async function sendRequest<T>(
    filePath: string,
    method: string,
    params: unknown
  ): Promise<T | undefined> {
    const ext = path.extname(filePath)
    const serverNames = extensionMap.get(ext)
    if (!serverNames?.length) return undefined

    const instance = servers.get(serverNames[0])
    if (!instance) return undefined

    if (!instance.isRunning()) await instance.start()

    return instance.sendRequest<T>(method, params)
  }

  return { initialize, shutdown, getServerForFile, sendRequest }
}
```

```typescript
// src/analytics/tokens.ts
export interface TokenUsage {
  inputTokens: number
  outputTokens: number
  cacheReadTokens: number
  cacheCreationTokens: number
  model: string
}

export class TokenTracker {
  private history: TokenUsage[] = []
  private maxEntries = 1000

  record(usage: TokenUsage): void {
    this.history.push({ ...usage, timestamp: Date.now() })
    if (this.history.length > this.maxEntries) {
      this.history = this.history.slice(-this.maxEntries)
    }
  }

  getTotalUsage(): TokenUsage {
    return this.history.reduce(
      (acc, u) => ({
        inputTokens: acc.inputTokens + u.inputTokens,
        outputTokens: acc.outputTokens + u.outputTokens,
        cacheReadTokens: acc.cacheReadTokens + u.cacheReadTokens,
        cacheCreationTokens: acc.cacheCreationTokens + u.cacheCreationTokens,
      }),
      { inputTokens: 0, outputTokens: 0, cacheReadTokens: 0, cacheCreationTokens: 0, model: '' }
    )
  }

  getCacheHitRate(): number {
    const total = this.getTotalUsage()
    const cached = total.cacheReadTokens + total.cacheCreationTokens
    const all = total.inputTokens + total.outputTokens + cached
    return all > 0 ? (cached / all) * 100 : 0
  }
}
```

### 9. Cost Tracker 实现

**OpenClaude 参考**: `src/cost-tracker.ts`

```typescript
// src/analytics/cost.ts
export interface ModelPricing {
  inputPerToken: number
  outputPerToken: number
  cacheReadPerToken: number
}

const MODEL_PRICING: Record<string, ModelPricing> = {
  'claude-opus-4-6': { inputPerToken: 0.000015, outputPerToken: 0.000075, cacheReadPerToken: 0.000001875 },
  'claude-sonnet-4-6': { inputPerToken: 0.000003, outputPerToken: 0.000015, cacheReadPerToken: 0.00000075 },
  'claude-haiku-4-5': { inputPerToken: 0.000001, outputPerToken: 0.000005, cacheReadPerToken: 0.00000025 },
  // 国产模型定价...
}

export function calculateCost(usage: TokenUsage): number {
  const pricing = MODEL_PRICING[usage.model] || MODEL_PRICING['claude-sonnet-4-6']

  const inputCost = usage.inputTokens * pricing.inputPerToken
  const outputCost = usage.outputTokens * pricing.outputPerToken
  const cacheCost = usage.cacheReadTokens * pricing.cacheReadPerToken

  return inputCost + outputCost + cacheCost
}

export class CostTracker {
  private totalCost: number = 0
  private modelUsage: Map<string, { tokens: TokenUsage; cost: number }> = new Map()

  addUsage(usage: TokenUsage): void {
    const cost = calculateCost(usage)
    this.totalCost += cost

    const existing = this.modelUsage.get(usage.model) || { tokens: { ...zeros }, cost: 0 }
    this.modelUsage.set(usage.model, {
      tokens: mergeTokenUsage(existing.tokens, usage),
      cost: existing.cost + cost,
    })
  }

  getTotalCost(): number {
    return this.totalCost
  }

  getCostByModel(): Record<string, number> {
    return Object.fromEntries(
      Array.from(this.modelUsage.entries()).map(([model, data]) => [model, data.cost])
    )
  }

  formatCost(): string {
    const cost = this.getTotalCost()
    return cost > 0.5 ? `$${cost.toFixed(2)}` : `$${cost.toFixed(4)}`
  }
}
```
