```typescript
// src/utils/hooks.ts - Hook 执行核心
async function executeHook(
  matcher: HookCallbackMatcher,
  input: HookInput,
  toolUseID: string | null,
  abort: AbortSignal
): Promise<HookResult> {
  const hook = matcher.hooks[0]
  const timeout = (hook.timeout || 60) * 1000

  // 1. 创建超时 AbortController
  const timeoutController = new AbortController()
  const combinedSignal = createCombinedAbortSignal(abort, timeoutController.signal)

  // 2. 根据 Hook 类型执行
  try {
    if (hook.type === 'command') {
      const result = await execCommandHook(hook, input, combinedSignal)
      return parseHookOutput(result)
    }
    if (hook.type === 'callback') {
      const result = await hook.callback(input, toolUseID, combinedSignal)
      return parseHookOutput(result)
    }
    // HTTP Hook / Agent Hook 类型...
  } catch (error) {
    if (error instanceof AbortError) {
      return { outcome: 'cancelled', preventContinuation: true }
    }
    return { outcome: 'non_blocking_error', blockingError: error.message }
  }
}

// Hook 结果解析
function parseHookOutput(output: HookJSONOutput): HookResult {
  if (isAsyncHookJSONOutput(output)) {
    registerPendingAsyncHook(output)
    return { outcome: 'success' }  // Async Hook 立即返回
  }

  const sync = output as SyncHookJSONOutput
  return {
    outcome: sync.decision === 'block' ? 'blocking' : 'success',
    preventContinuation: sync.continue === false,
    permissionBehavior: sync.hookSpecificOutput?.permissionDecision,
    updatedInput: sync.hookSpecificOutput?.updatedInput,
    additionalContext: sync.hookSpecificOutput?.additionalContext,
  }
}
```

```typescript
// src/utils/hookChains.ts
export async function executeHookChain(
  event: HookEvent,
  matchers: HookCallbackMatcher[],
  input: HookInput,
  abort: AbortSignal
): Promise<AggregatedHookResult> {
  const results: HookResult[] = []

  for (const matcher of matchers) {
    // 检查 matcher 是否匹配
    if (!matchesTool(matcher.matcher, input)) continue

    const result = await executeHook(matcher, input, abort)
    results.push(result)

    // 阻塞 Hook 停止后续执行
    if (result.preventContinuation) break
  }

  return aggregateHookResults(results)
}

function aggregateHookResults(results: HookResult[]): AggregatedHookResult {
  // 找到第一个 blocking 结果
  const blocking = results.find(r => r.outcome === 'blocking')

  return {
    outcome: blocking ? 'blocking' : 'success',
    preventContinuation: results.some(r => r.preventContinuation),
    permissionBehavior: results.find(r => r.permissionBehavior)?.permissionBehavior,
    updatedInput: results.find(r => r.updatedInput)?.updatedInput,
    additionalContexts: results.filter(r => r.additionalContext).map(r => r.additionalContext),
    blockingErrors: results.filter(r => r.blockingError).map(r => r.blockingError!)
  }
}
```

```typescript
// src/tools/AgentTool/runAgent.ts
async function initializeAgentMcpServers(
  agentDefinition: AgentDefinition,
  parentClients: MCPServerConnection[]
): Promise<{
  clients: MCPServerConnection[]
  tools: Tools
  cleanup: () => Promise<void>
}> {
  if (!agentDefinition.mcpServers?.length) {
    return { clients: parentClients, tools: [], cleanup: async () => {} }
  }

  const agentClients: MCPServerConnection[] = []
  const newlyCreatedClients: MCPServerConnection[] = []
  const agentTools: Tool[] = []

  for (const spec of agentDefinition.mcpServers) {
    // 字符串引用 vs 内联定义
    if (typeof spec === 'string') {
      // 引用已存在的 MCP 配置
      const config = getMcpConfigByName(spec)
      const client = await connectToServer(spec, config)
      agentClients.push(client)
    } else {
      // 内联定义 { name: config }
      const [name, config] = Object.entries(spec)[0]
      const scopedConfig = { ...config, scope: 'dynamic' }
      const client = await connectToServer(name, scopedConfig)
      agentClients.push(client)
      newlyCreatedClients.push(client)  // 需要清理
    }

    // 获取工具列表
    if (client.type === 'connected') {
      const tools = await fetchToolsForClient(client)
      agentTools.push(...tools)
    }
  }

  // 清理函数 - 只清理新创建的客户端
  const cleanup = async () => {
    for (const client of newlyCreatedClients) {
      if (client.type === 'connected') {
        await client.cleanup()
      }
    }
  }

  return { clients: [...parentClients, ...agentClients], tools: agentTools, cleanup }
}
```

```typescript
  // 1. 初始化 Agent MCP
  const { clients, tools, cleanup } = await initializeAgentMcpServers(
    agentDefinition,
    toolUseContext.options.mcpClients
  )

  // 2. 创建子 Agent Context
  const subContext = createSubagentContext(toolUseContext, agentDefinition)
  subContext.options.mcpClients = clients
  subContext.options.tools = [...toolUseContext.options.tools, ...tools]

  // 3. 执行 Agent Query 循环
  try {
    for await (const message of query(subContext)) {
      yield message
      if (agentDefinition.maxTurns && turnCount >= agentDefinition.maxTurns) break
    }
  } finally {
    // 4. 清理 Agent MCP 资源
    await cleanup()
    clearSessionHooks()
    cleanupAgentTracking()
    killShellTasksForAgent()
  }
}
```

```typescript
// src/utils/toolResultStorage.ts
const TOOL_RESULT_BUDGET = 500_000  // 总预算 500KB

function applyToolResultBudget(
  messages: Message[],
  state: ContentReplacementState
): Message[] {
  let totalChars = 0
  const replacements: Map<string, string> = new Map()

  for (const msg of messages) {
    if (msg.type === 'tool_result') {
      const content = msg.content as string
      if (content.length + totalChars > TOOL_RESULT_BUDGET) {
        // 持久化超出预算的结果
        const filePath = persistToolResult(msg.tool_use_id, content)
        replacements.set(msg.tool_use_id, filePath)
        msg.content = filePath  // 替换为文件路径
      } else {
        totalChars += content.length
      }
    }
  }

  state.replacements = replacements
  state.totalReplacedChars = totalChars
  return messages
}
```

#### 5. Query 执行流程

**核心文件**: `src/query.ts`

```typescript
// src/query.ts - 主执行流程
export async function* query(context: QueryContext): AsyncGenerator<Message> {
  const { messages, tools, mcpClients } = context

  // 1. 构建系统提示词
  const systemPrompt = await buildSystemPrompt(context)

  // 2. 预加载内存和技能
  await preloadMemoryAndSkills(context)

  // 3. 执行 SessionStart Hooks
  await executeSessionStartHooks(context)

  // 4. 主循环
  while (true) {
    // 4.1 调用 LLM API
    const response = await callAPI({
      model: context.model,
      system: systemPrompt,
      messages: normalizeMessagesForAPI(messages),
      tools: tools.filter(t => t.isEnabled())
    })

    // 4.2 处理响应
    for (const block of response.content) {
      if (block.type === 'text') {
        yield createAssistantMessage({ content: block.text })
      }
      if (block.type === 'tool_use') {
        yield createToolUseMessage(block)

        // 4.3 执行 PreToolUse Hooks
        const hookResult = await executePreToolUseHooks(block.name, block.input)

        if (hookResult.preventContinuation) {
          yield createHookBlockMessage(hookResult)
          continue
        }

        // 4.4 执行工具
        const tool = findToolByName(tools, block.name)
        const toolResult = await tool.call(block.input, context)

        // 4.5 持久化大结果
        if (toolResult.data.length > tool.maxResultSizeChars) {
          const filePath = await persistToolResult(block.id, toolResult.data)
          toolResult.data = filePath
        }

        // 4.6 执行 PostToolUse Hooks
        await executePostToolUseHooks(block.name, block.input, toolResult)

        yield createToolResultMessage(block.id, toolResult)
      }
    }

    // 4.7 检查 Compact 条件
    if (shouldCompact(messages)) {
      await executePreCompactHooks()
      messages = await compactMessages(messages)
      await executePostCompactHooks()
    }

    // 4.8 检查终止条件
    if (response.stop_reason === 'end_turn') break
    if (context.maxTurns && turnCount >= context.maxTurns) break
  }

  // 5. 执行 SessionEnd Hooks
  await executeSessionEndHooks(context)
}
```

```typescript
export async function execCommandHook(
  hook: HookCommand,
  input: HookInput,
  abort: AbortSignal
): Promise<HookJSONOutput> {
  // 构建 Hook 输入环境变量
  const env = {
    HOOK_EVENT: input.event || 'PreToolUse',
    HOOK_TOOL_NAME: input.tool_name || '',
    HOOK_TOOL_INPUT: JSON.stringify(input.tool_input || {}),
    HOOK_SESSION_ID: getSessionId(),
    HOOK_PROJECT_ROOT: getCwd(),
  }

  // 执行 shell 命令
  const result = await exec(hook.command, {
    env: { ...process.env, ...env },
    signal: abort,
    timeout: (hook.timeout || 60) * 1000,
    shell: hook.shell || 'bash'
  })

  // 解析 JSON 输出
  try {
    const output = JSON.parse(result.stdout)
    return output as HookJSONOutput
  } catch {
    // 非 JSON 输出 = 成功继续
    return { continue: true }
  }
}

export async function execCallbackHook(
  hook: HookCallback,
  input: HookInput,
  abort: AbortSignal
): Promise<HookJSONOutput> {
  const timeout = (hook.timeout || 60) * 1000

  // 创建超时 AbortController
  const timeoutController = new AbortController()
  setTimeout(() => timeoutController.abort(), timeout)

  const combinedSignal = AbortSignal.any([abort, timeoutController.signal])

  try {
    const result = await hook.callback(input, combinedSignal)
    return result
  } catch (error) {
    if (error.name === 'AbortError') {
      return { continue: false, stopReason: 'Hook timeout' }
    }
    return { continue: true }  // non_blocking
  }
}
```

```typescript
// src/hooks/chain.ts
export async function executeHookChain(
  event: HookEvent,
  matchers: HookMatcher[],
  input: HookInput,
  abort: AbortSignal
): Promise<AggregatedHookResult> {
  const results: HookResult[] = []

  for (const matcher of matchers) {
    // 检查 matcher 是否匹配工具
    if (input.tool_name && matcher.matcher) {
      if (!matchesTool(matcher.matcher, input.tool_name, input.tool_input || {})) {
        continue
      }
    }

    // 执行 Hook
    const result = await executeHook(matcher, input, event, abort)
    results.push(result)

    // 阻塞 Hook 停止后续执行
    if (result.preventContinuation) {
      break
    }
  }

  return aggregateHookResults(results)
}

function aggregateHookResults(results: HookResult[]): AggregatedHookResult {
  const blocking = results.find(r => r.outcome === 'blocking')
  const blockingErrors = results.filter(r => r.blockingError).map(r => ({
    blockingError: r.blockingError!,
    command: r.command || ''
  }))

  return {
    outcome: blocking ? 'blocking' : 'success',
    preventContinuation: results.some(r => r.preventContinuation),
    permissionBehavior: results.find(r => r.permissionBehavior)?.permissionBehavior,
    updatedInput: results.find(r => r.updatedInput)?.updatedInput,
    additionalContexts: results.filter(r => r.additionalContext).map(r => r.additionalContext!),
    blockingErrors,
    stopReason: blocking?.stopReason
  }
}
```

```typescript
async function executeToolWithHooks(
  tool: OpenHorseTool,
  toolUseId: string,
  args: Record<string, unknown>,
  context: ToolContext
): Promise<ToolResult> {
  const matchers = getActiveHooks('PreToolUse')

  // 1. 执行 PreToolUse Hooks
  const preResult = await executeHookChain(
    'PreToolUse',
    matchers,
    { tool_name: tool.name, tool_input: args },
    context.abortSignal

    ...context,
    additionalContext: preResult.additionalContexts?.join('\n')

  const postMatchers = getActiveHooks('PostToolUse')
  await executeHookChain(
    'PostToolUse',
    postMatchers,
    { tool_name: tool.name, tool_input: effectiveArgs, tool_result: result },
    context.abortSignal
  )

  return result
}
```

```typescript
        continue
      }
      const client = await connectToMcpServer(spec, config)
      agentClients.push(client)
    } else {
      // 内联定义
      const [name, config] = Object.entries(spec)[0]
      const scopedConfig = { ...config, scope: 'agent' }
      const client = await connectToMcpServer(name, scopedConfig)
      agentClients.push(client)
      newlyCreatedClients.push(client)  // 需要清理
    }

    // 获取工具列表
    if (client.status === 'connected') {
      const tools = await fetchMcpTools(client)
      agentTools.push(...tools)
    }
  }

  // 清理函数
  const cleanup = async () => {
    for (const client of newlyCreatedClients) {
      if (client.status === 'connected') {
        await client.disconnect()
      }
    }
  }

  return {
    clients: [...parentClients, ...agentClients],
    tools: agentTools,
    cleanup
  }
}
```

```typescript
// src/services/tool-result-storage.ts
const TOOL_RESULT_DIR = '.openhorse/tool-results'

export interface ContentReplacementState {
  replacements: Map<string, string>
  totalReplacedChars: number
}

export function shouldPersistToolResult(result: string, maxChars: number): boolean {
  return result.length > maxChars
}

export async function persistToolResult(
  toolUseId: string,
  result: string
): Promise<string> {
  const filePath = getToolResultPath(toolUseId)
  await ensureDir(dirname(filePath))
  await writeFile(filePath, result)

  return `Tool result persisted to: ${filePath}\nUse Read tool to access full content.`
}

export function getToolResultPath(toolUseId: string): string {
  return join(getSessionDir(), TOOL_RESULT_DIR, `${toolUseId}.txt`)
}

export function applyToolResultBudget(
  messages: Message[],
  budget: number = 500_000
): Message[] {
  let totalChars = 0

  for (const msg of messages) {
    if (msg.type === 'tool_result' && typeof msg.content === 'string') {
      if (msg.content.length + totalChars > budget) {
        // 标记需要持久化
        msg.persisted = true
      } else {
        totalChars += msg.content.length
      }
    }
  }

  return messages
}
```
