```typescript
export class StdioTransport implements MCPTransport {
  private process: ChildProcess;
  private client: Client;

  async connect(config: McpStdioServerConfig): Promise<void> {
    // 1. 启动子进程
    this.process = spawn(config.command, config.args ?? [], {
      env: { ...process.env, ...config.env },
      stdio: ['pipe', 'pipe', 'pipe'],
    });

    // 2. 创建 MCP Client
    this.client = new Client(
      { name: 'openhorse-mcp', version: '0.1.0' },
      { capabilities: {} }
    );

    // 3. 连接 transport
    const transport = new StdioClientTransport(
      this.process.stdin,
      this.process.stdout
    );
    await this.client.connect(transport);
  }

  async listTools(): Promise<Tool[]> {
    const { tools } = await this.client.listTools();
    return tools;
  }

  async callTool(name: string, args: Record<string, unknown>): Promise<ToolResult> {
    const result = await this.client.callTool({ name, arguments: args });
    return result;
  }

  async cleanup(): Promise<void> {
    await this.client.close();
    this.process.kill();
  }
}

export class HTTPTransport implements MCPTransport {
  private client: Client;

    const transport = new HTTPClientTransport(
      new URL(config.url),
      { headers: config.headers }
    );
```

```typescript
// src/mcp/client-manager.ts
export class MCPClientManager {
  private connections: Map<string, MCPServerConnection> = new Map();
  private tools: Map<string, MCPToolWrapper> = new Map();

  async initialize(configs: Record<string, ScopedMcpServerConfig>): Promise<void> {
    for (const [name, config] of Object.entries(configs)) {
      await this.connectServer(name, config);
    }
  }

  async connectServer(name: string, config: ScopedMcpServerConfig): Promise<void> {
    try {
      // 1. 创建 Transport
      const transport = createTransport(config);

      // 2. 连接
      await transport.connect(config);

      // 3. 获取 capabilities
      const capabilities = await transport.getCapabilities();

      // 4. 注册工具
      if (capabilities.tools) {
        const tools = await transport.listTools();
        for (const tool of tools) {
          this.registerTool(name, tool, transport);
        }
      }

      // 5. 更新状态
      this.connections.set(name, {
        type: 'connected',
        name,
        client: transport.client,
        capabilities,
        config,
        cleanup: transport.cleanup,
      });

    } catch (error) {
      this.connections.set(name, {
        type: 'failed',
        name,
        config,
        error: error.message,
      });
    }
  }

  registerTool(serverName: string, tool: Tool, transport: MCPTransport): void {
    const wrappedTool = wrapMCPTool(serverName, tool, transport);
    // 工具名格式: mcp__{serverName}__{toolName}
    const toolId = `mcp__${serverName}__${tool.name}`;
    this.tools.set(toolId, wrappedTool);
  }

  getTools(): Tool[] {
    return Array.from(this.tools.values());
  }

  async cleanup(): Promise<void> {
    for (const conn of this.connections.values()) {
      if (conn.type === 'connected') {
        await conn.cleanup();
      }
    }
    this.connections.clear();
    this.tools.clear();
  }
}
```

```typescript
      });
    }
  }

  // 2. 加载 agents
  if (plugin.agentsPath) {
    const agentFiles = await glob(`${plugin.agentsPath}/*.md`);
    for (const file of agentFiles) {
      const agent = await parseMarkdownAgent(file);
      extensions.agents.push({
        ...agent,
        source: plugin.source,
      });
    }
  }

  return extensions;
}
```

```typescript
// src/plugins/directory-loader.ts
export async function loadPluginsFromDirectory(dir: string): Promise<PluginLoadResult> {
  const enabled: LoadedPlugin[] = [];
  const errors: PluginError[] = [];

  const entries = await readdir(dir, { withFileTypes: true });

  for (const entry of entries) {
    if (!entry.isDirectory()) continue;

    const pluginPath = join(dir, entry.name);

    try {
      const plugin = await loadPluginFromDirectory(pluginPath);
      if (plugin) {
        enabled.push(plugin);
      }
    } catch (error) {
      errors.push({
        type: 'generic-error',
        source: entry.name,
        error: error.message,
      });
    }
  }

  return { enabled, disabled: [], errors };
}
```

## OpenClaude vs OpenHorse 对比分析 (v0.1.18 视角)

### 可引入的关键设计

| OpenClaude 设计 | OpenHorse v0.1.18 实现 | 简化点 |
|----------------|----------------------|-------|
| **8 种 Transport** | 4 种 (stdio/sse/http/sdk) | 去掉 IDE 内部类型、claudeai-proxy |
| **MCPServerConnection 5 状态** | 5 状态 (完整借鉴) | 保持设计 |
| **ConfigScope 7 种** | 3 种 (user/project/local) | 去掉 enterprise/managed/claudeai/dynamic |
| **Policy 过滤 (allow/deny list)** | 可选 (P2) | 企业场景简化 |
| **Plugin Dedup 签名机制** | 简化版 | 基于名称去重，不做签名匹配 |
| **PluginError 20+ 种** | 核心 10 种 | 去掉 mcpb-download/extract、lsp 相关 |
| **PluginManifest 15 字段** | 核心 8 字段 | 去掉 lspServers、userConfig、homepage、license |
| **Marketplace 系统** | 无 (P3) | 仅支持本地目录加载 |
| **Versioned Cache** | 简化版 | 无版本化缓存，直接目录 |

### OpenHorse 差异化竞争优势

| 方向 | OpenClaude | OpenHorse v0.1.18 |
|------|-----------|------------------|
| **Transport 类型** | 8 种 (含 IDE 内部) | 4 种核心类型 |
| **Config Scope** | 7 种 (含 enterprise/managed) | 3 种简化 scope |
| **Policy 过滤** | 企业级 allow/deny list | 可选，默认无限制 |
| **Plugin Dedup** | 签名匹配 (复杂) | 名称匹配 (简单) |
| **Marketplace** | 完整 marketplace 系统 | 仅本地目录 (--plugin-dir) |
| **Plugin Cache** | 版本化 ZIP 缓存 | 直接目录加载 |
| **学习曲线** | 高 (Marketplace, Policy, Dedup) | 低 (简单目录加载) |

**核心竞争力**:
1. **简化 MCP Transport** — 4 种核心类型覆盖 95% 使用场景
2. **三 Scope 配置** — user/project/local，无企业复杂性
3. **目录 Plugin** — --plugin-dir 直接加载，无 marketplace 依赖
4. **零 Policy 依赖** — 默认无限制，适合个人开发者
5. **MCP Tool 集成** — mcp__server__tool 命名空间，无缝融入
