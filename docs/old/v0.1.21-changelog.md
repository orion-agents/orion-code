```typescript
// src/utils/hooks.ts
async function executeHook(
  matcher: HookCallbackMatcher,
  input: HookInput,
  toolUseID: string | null,
  abort: AbortSignal
): Promise<HookResult> {
  const hook = matcher.hooks[0]
  const timeout = (hook.timeout || 60) * 1000

  // 1. 创建超时 AbortController
  const timeoutController = new AbortController()
  const combinedSignal = createCombinedAbortSignal(abort, timeoutController.signal)

  // 2. 执行 Hook
  try {
    const result = await hook.callback(input, toolUseID, combinedSignal)
    
    // 3. 处理输出
    if (isAsyncHookJSONOutput(result)) {
      // 注册异步 Hook
      registerPendingAsyncHook(...)
      return { outcome: 'success' }
    }
    
    // 4. 解析 Sync 输出
    return parseSyncHookResult(result)
    
  } catch (error) {
    if (error instanceof AbortError) {
      return { outcome: 'cancelled', preventContinuation: true }
    }
    return { outcome: 'non_blocking_error', blockingError: error.message }
  }
}

// Hook Chain 执行
async function executeHookChain(
  event: HookEvent,
  matchers: HookCallbackMatcher[],
  input: HookInput
): Promise<AggregatedHookResult> {
  const results: HookResult[] = []
  
  for (const matcher of matchers) {
    const result = await executeHook(matcher, input, ...)
    results.push(result)
    
    // 阻塞 Hook 停止后续执行
    if (result.preventContinuation) break
  }
  
  return aggregateHookResults(results)
}
```

async function* runAgent(params: RunAgentParams): AsyncGenerator<Message> {
  const { agentDefinition, promptMessages, toolUseContext } = params
  
  // 1. 解析 Agent 模型
  const model = resolveAgentModel(agentDefinition, toolUseContext)
  
  // 2. 解析 Agent Provider (路由覆盖)
  const provider = resolveAgentProvider(agentDefinition, toolUseContext)
  
  // 3. 初始化 Agent 专属 MCP
  if (agentDefinition.mcpServers) {
    await initializeAgentMcpServers(agentDefinition.mcpServers, toolUseContext)
  }
  
  // 4. 解析 Agent 工具集
  const tools = resolveAgentTools(agentDefinition, toolUseContext)
  
  // 5. 构建 Agent System Prompt
  const systemPrompt = getAgentSystemPrompt(agentDefinition, toolUseContext)
  
  // 6. 执行 SubagentStart Hooks
  await executeSubagentStartHooks(agentDefinition.agentType, promptMessages)
  
  // 7. 注册 Agent Hooks
  if (agentDefinition.hooks) {
    registerFrontmatterHooks(agentDefinition.hooks)
  }
  
  // 8. 预加载技能
  if (agentDefinition.skills) {
    await preloadSkills(agentDefinition.skills)
  }
  
  // 9. 创建子 Agent Context
  const subContext = createSubagentContext(toolUseContext, agentDefinition)
  
  // 10. 执行 Query 循环
  for await (const message of query(subContext)) {
    yield message
    
    // 检查 maxTurns 限制
    if (agentDefinition.maxTurns && turnCount >= agentDefinition.maxTurns) {
      break
    }
  }
  
  // 11. 清理资源
  finally {
    await cleanupAgentMcpServers()
    clearSessionHooks()
    cleanupAgentTracking()
    agentReadFileState.clear()
    killShellTasksForAgent()
  }
}

// src/utils/permissions/permissionRuleParser.ts
function permissionRuleValueFromString(rule: string): PermissionRuleValue {
  // 格式: ToolName 或 ToolName(content)
  // 例: Bash → { toolName: 'Bash' }
  // 例: Bash(git *) → { toolName: 'Bash', ruleContent: 'git *' }
  
  const match = rule.match(/^([^(]+)(?:\(([^)]+)\))?$/)
  if (!match) throw new Error(`Invalid rule: ${rule}`)
  
  return {
    toolName: match[1],
    ruleContent: match[2]
  }
}

function permissionRuleValueToString(rule: PermissionRuleValue): string {
  if (rule.ruleContent) {
    return `${rule.toolName}(${rule.ruleContent})`
  }
  return rule.toolName
}

export function matchesTool(matcher: string, toolName: string, input: Record<string, unknown>): boolean {
  if (!matcher) return true  // 无 matcher = 匹配所有
  
  // 格式: ToolName(pattern)
  const match = matcher.match(/^([^(]+)(?:\(([^)]+)\))?$/)
  if (!match) return false
  
  const [, name, pattern] = match
  
  // 工具名匹配
  if (name !== toolName) return false
  
  // 模式匹配
  if (pattern) {
    // Bash: pattern 匹配命令
    // Read/Edit: pattern 匹配路径
    return matchPattern(pattern, input)
  }
  
  return true
}

function matchPattern(pattern: string, input: Record<string, unknown>): boolean {
  // 通配符匹配
  const regex = new RegExp(pattern.replace('*', '.*'))
  
  // Bash: 匹配 command
  if (input.command) return regex.test(input.command as string)
  
  // Read/Edit: 匹配 file_path
  if (input.file_path) return regex.test(input.file_path as string)
  
  return false
}

**目标**: 实现 Hook 执行流程

**技术方案**:

```typescript
// src/hooks/executor.ts
export async function executeHook(
  matcher: HookMatcher,
  input: HookInput,
  event: HookEvent,
  abort: AbortSignal
): Promise<HookResult> {
  const hook = matcher.hooks[0]
  const timeout = (hook.timeout || 60) * 1000
  
  // 创建超时 AbortController
  const timeoutController = new AbortController()
  setTimeout(() => timeoutController.abort(), timeout)
  
  // 合并 abort signals
  const combinedSignal = AbortSignal.any([abort, timeoutController.signal])
  
  try {
    // 执行命令 Hook
    if (hook.type === 'command' && hook.command) {
      const result = await executeCommandHook(hook.command, input, combinedSignal)
      return parseHookOutput(result)
    }
    
    // 执行回调 Hook
    if (hook.type === 'callback' && hook.callback) {
      const result = await hook.callback(input, combinedSignal)
      return parseHookOutput(result)
    }
    
    return { outcome: 'success' }
    
  } catch (error) {
    if (error.name === 'AbortError') {
      return { outcome: 'cancelled', preventContinuation: true }
    }
    return { outcome: 'non_blocking_error', message: error.message }
  }
}

async function executeCommandHook(
  command: string,
  input: HookInput,
  abort: AbortSignal
): Promise<HookOutput> {
  // 构建 Hook 输入环境变量
  const env = {
    HOOK_EVENT: input.event,
    HOOK_TOOL_NAME: input.tool_name,
    HOOK_TOOL_INPUT: JSON.stringify(input.tool_input || {}),
    // ...
  }
  
  // 执行命令
  const result = await exec(command, {
    env: { ...process.env, ...env },
    signal: abort,
    timeout: 60_000
  })
  
  // 解析 JSON 输出
  try {
    return JSON.parse(result.stdout)
  } catch {
    // 非 JSON 输出 = 成功继续
    return { continue: true }
  }
}

function parseHookOutput(output: HookOutput): HookResult {
  if ('async' in output && output.async) {
    return { outcome: 'success' }  // 异步 Hook 立即返回
  }
  
  const sync = output as SyncHookOutput
  
  return {
    outcome: sync.decision === 'block' ? 'blocking' : 'success',
    preventContinuation: sync.continue === false,
    permissionBehavior: sync.permissionDecision,
    updatedInput: sync.updatedInput,
    message: sync.systemMessage,
  }
}

// src/hooks/chain.ts
export async function executeHookChain(
  event: HookEvent,
  config: HooksConfig,
  input: HookInput,
  abort: AbortSignal
): Promise<AggregatedHookResult> {
  const matchers = config[event] || []
  const results: HookResult[] = []
  
  for (const matcher of matchers) {
    // 检查 matcher 是否匹配
    if (input.tool_name && !matchesTool(matcher.matcher || '', input.tool_name, input.tool_input || {})) {
      continue
    }
    
    // 执行 Hook
    const result = await executeHook(matcher, input, event, abort)
    results.push(result)
    
    // 阻塞 Hook 停止后续执行
    if (result.preventContinuation) {
      break
    }
  }
  
  // 聚合结果
  return aggregateResults(results)
}

function aggregateResults(results: HookResult[]): AggregatedHookResult {
  const blocking = results.find(r => r.outcome === 'blocking')
  
  return {
    outcome: blocking ? 'blocking' : 'success',
    preventContinuation: results.some(r => r.preventContinuation),
    permissionBehavior: results.find(r => r.permissionBehavior)?.permissionBehavior,
    updatedInput: results.find(r => r.updatedInput)?.updatedInput,
    messages: results.filter(r => r.message).map(r => r.message),
  }
}
```

// src/permissions/matcher.ts
export function checkPermissionRules(
  toolName: string,
  input: Record<string, unknown>,
  rules: PermissionRulesConfig
): PermissionResult {
  // 1. 检查 deny 规则 (最高优先级)
  for (const source of ['session', 'projectSettings', 'userSettings']) {
    const denyRules = rules[source]?.deny || []
    for (const rule of denyRules) {
      if (matchesRule(rule, toolName, input)) {
        return {
          behavior: 'deny',
          reason: `Denied by rule: ${rule} (${source})`
        }
      }
    }
  }
  
  // 2. 检查 allow 规则
  for (const source of ['session', 'projectSettings', 'userSettings']) {
    const allowRules = rules[source]?.allow || []
    for (const rule of allowRules) {
      if (matchesRule(rule, toolName, input)) {
        return { behavior: 'allow' }
      }
    }
  }
  
  // 3. 检查 ask 规则
  for (const source of ['session', 'projectSettings', 'userSettings']) {
    const askRules = rules[source]?.ask || []
    for (const rule of askRules) {
      if (matchesRule(rule, toolName, input)) {
        return {
          behavior: 'ask',
          reason: `Ask required by rule: ${rule} (${source})`
        }
      }
    }
  }
  
  // 4. 默认: 根据模式决定
  return { behavior: 'ask' }
}

function matchesRule(
  rule: string,
  toolName: string,
  input: Record<string, unknown>
): boolean {
  const value = parsePermissionRule(rule)
  
  // 工具名匹配
  if (value.toolName !== toolName) return false
  
  // 规则内容匹配
  if (value.ruleContent) {
    return matchesRuleContent(value.ruleContent, toolName, input)
  }
  
  return true
}

function matchesRuleContent(
  content: string,
  toolName: string,
  input: Record<string, unknown>
): boolean {
  // Bash: 匹配命令
  if (toolName === 'Bash' && input.command) {
    const regex = new RegExp(content.replace('*', '.*'))
    return regex.test(input.command as string)
  }
  
  // Read/Edit/Write: 匹配路径
  if (['Read', 'Edit', 'Write'].includes(toolName) && input.file_path) {
    const regex = new RegExp(content.replace('*', '.*'))
    return regex.test(input.file_path as string)
  }
  
  return false
}
