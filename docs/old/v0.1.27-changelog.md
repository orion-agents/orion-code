export function deserializeMessagesWithInterruptDetection(
  entries: SessionLogEntry[],
): {
  messages: Message[]
  turnInterruptionState: TurnInterruptionState
} {
  let turnInterruptionState: TurnInterruptionState = 'none'
  const messages: Message[] = []
  
  for (const entry of entries) {
    if (entry.type === 'assistant') {
      // 3P providers 需要移除 thinking blocks
      const shouldStripThinking = !isFirstPartyAnthropicBaseUrl(getAPIProvider())
      
      if (shouldStripThinking) {
        const stripped = stripThinkingBlocks(entry.message.content)
        messages.push({
          ...entry,
          message: { ...entry.message, content: stripped }
        })
      } else {
        messages.push(entry)
      }
      
      // 检测中断状态
      if (entry.message.stop_reason === 'stop_sequence_interrupted') {
        turnInterruptionState = 'interrupted_turn'
      }
    }
    
    if (entry.type === 'user') {
      // Filter unresolved tool uses
      const filtered = filterUnresolvedToolUses(entry.message.content)
      messages.push({ ...entry, message: { ...entry.message, content: filtered } })
    }
  }
  
  return { messages, turnInterruptionState }
}
```

export async function loadConversationForResume(
  sessionId: string,
  options?: {
    entrypoint?: ResumeEntrypoint
    skipCompactReplay?: boolean
  },
): Promise<{
  messages: Message[]
  turnInterruptionState: TurnInterruptionState
  sessionId: string
}> {
  const sessionLog = await loadSessionLog(sessionId)
  
  // Size validation
  const totalBytes = estimateMessageBytes(sessionLog)
  if (totalBytes > MAX_RESUME_MESSAGE_BYTES) {
    throw new Error(`Session too large to resume (${totalBytes} > ${MAX_RESUME_MESSAGE_BYTES})`)

    const replayed = await replayCompactBoundary(messages)
    return { messages: replayed, turnInterruptionState, sessionId }
  }
  
  return { messages, turnInterruptionState, sessionId }

      throw error
    }

    return this.clients.get(name)
  }
  
  getInitializationState(): InitializationState {
    return this.initializationState
  }

  getGeneration(): number {
    return this.generation
  }
}

export class QueryEngine {
  private config: QueryEngineConfig
  private mutableMessages: Message[] = []
  private totalUsage: NonNullableUsage = EMPTY_USAGE
  private permissionDenials: PermissionDenial[] = []
  private readFileState: FileStateCache = new Map()
  private abortController: AbortController
  private loadedNestedMemoryPaths: Set<string> = new Set()
  private discoveredSkillNames: Set<string> = new Set()
  private hasHandledOrphanedPermission = false
  
  constructor(config: QueryEngineConfig) {
    this.config = config
    this.abortController = config.abortController ?? new AbortController()
    this.mutableMessages = config.initialMessages ?? []
    this.readFileState = config.readFileCache ?? new Map()
  }
  
  async *submitMessage(
    prompt: string | Array<ContentBlockParam>,
    options?: { uuid?: string; isMeta?: boolean },
  ): AsyncGenerator<SDKMessage, void, unknown> {
    // 1. Process user input (slash commands, attachments)
    const { messagesFromUserInput, shouldQuery, model, resultText } = 
      await processUserInput({
        input: prompt,
        mode: 'prompt',
        messages: this.mutableMessages,
        context: processUserInputContext,
      })
    
    // 2. Push messages to mutableMessages
    this.mutableMessages.push(...messagesFromUserInput)
    
    // 3. Persist transcript (async for bare mode)
    if (persistSession && messagesFromUserInput.length > 0) {
      const transcriptPromise = recordTranscript(this.mutableMessages)
      if (isBareMode()) {
        void transcriptPromise
      } else {
        await transcriptPromise
      }
    }
    
    // 4. Build system init message
    yield buildSystemInitMessage({
      tools,
      mcpClients,
      model: mainLoopModel,
      permissionMode,
      commands,
      agents,
      skills,
      plugins,
    })
    
    // 5. If shouldQuery, enter query loop
    if (shouldQuery) {
      for await (const message of query({
        messages,
        systemPrompt,
        userContext,
        systemContext,
        canUseTool,
        toolUseContext,
        querySource: 'sdk',
        maxTurns,
        taskBudget,
      })) {
        // Handle message types
        switch (message.type) {
          case 'assistant':
            this.mutableMessages.push(message)
            yield* normalizeMessage(message)
            break
          case 'user':
            this.mutableMessages.push(message)
            yield* normalizeMessage(message)
            break
          case 'progress':
            this.mutableMessages.push(message)
            if (persistSession) void recordTranscript(messages)
            yield* normalizeMessage(message)
            break
          case 'stream_event':
            // Track usage
            if (message.event.type === 'message_start') {
              currentMessageUsage = updateUsage(currentMessageUsage, message.event.message.usage)
            }
            if (message.event.type === 'message_delta') {
              lastStopReason = message.event.delta.stop_reason
            }
            if (message.event.type === 'message_stop') {
              this.totalUsage = accumulateUsage(this.totalUsage, currentMessageUsage)
            }
            if (includePartialMessages) yield { type: 'stream_event', ... }
            break
          case 'attachment':
            // Handle structured_output, max_turns_reached, queued_command
            break
          case 'system':
            // Handle compact_boundary, api_error
            break
        }
        
        // Budget checks
        if (maxBudgetUsd !== undefined && getTotalCost() >= maxBudgetUsd) {
          yield { type: 'result', subtype: 'error_max_budget_usd', ... }
          return
        }
      }
    }
    
    // 6. Yield final result
    yield {
      type: 'result',
      subtype: 'success',
      result: textResult,
      usage: this.totalUsage,
      ...
    }
  }
  
  // Engine management methods
  interrupt(): void {
    this.abortController.abort()
  }
  
  getMessages(): readonly Message[] {
    return this.mutableMessages
  }
  
  injectMessages(messages: Message[]): void {
    // Validation + push
    const validated = validateArrayOf(messages, ...)
    this.mutableMessages.push(...validated)
  }
  
  injectAgents(agents: AgentDefinition[]): void {
    // Validate agentType, whenToUse, getSystemPrompt
    this.config.agents = validated
  }
  
  updateTools(tools: Tools): void {
    // Transactional: validate agents compatibility before commit
    this.config.tools = toolArray
    invalidateRemovedToolSchemas(validToolNames)
  }
  
  setThinkingConfig(config: ThinkingConfig): void {
    this.config.thinkingConfig = config
  }
  
  setMcpClients(clients: MCPServerConnection[]): void {
    this.config.mcpClients = clients
  }
}
```

#### 6. Streaming Token Counter 实时计数架构

**核心文件**: `src/utils/streamingTokenCounter.ts` (133 行)

**StreamingTokenCounter 类**:
```typescript
export class StreamingTokenCounter {
  private inputTokens = 0
  private accumulatedContent = ''
  private lastCountedIndex = 0
  private cachedOutputTokens = 0
  private startTime = 0
  
  start(initialInputTokens?: number): void {
    this.reset()
    this.startTime = Date.now()
    this.inputTokens = initialInputTokens ?? 0
  }
  
  addChunk(deltaContent?: string): void {
    if (deltaContent) {
      this.accumulatedContent += deltaContent
      this.recountAtWordBoundary()  // Only count at word boundaries
    }
  }
  
  private recountAtWordBoundary(): void {
    const unprocessedContent = this.accumulatedContent.slice(this.lastCountedIndex)
    
    // Find next space for word boundary
    const nextSpaceIndex = unprocessedContent.indexOf(' ')
    
    const shouldCount =
      nextSpaceIndex > 0 ||
      unprocessedContent.length > 50 ||
      unprocessedContent.length === 0
    
    if (shouldCount) {
      const boundaryIndex = this.lastCountedIndex + (nextSpaceIndex > 0 ? nextSpaceIndex : this.accumulatedContent.length)
      this.cachedOutputTokens = roughTokenCountEstimation(this.accumulatedContent.slice(0, boundaryIndex))
      this.lastCountedIndex = boundaryIndex
    }
  }
  
  finalize(): number {
    if (this.accumulatedContent.length > this.lastCountedIndex) {
      this.cachedOutputTokens = roughTokenCountEstimation(this.accumulatedContent)
    }
    return this.cachedOutputTokens
  }
  
  // Computed properties
  get total(): number { return this.inputTokens + this.cachedOutputTokens }
  get output(): number { return this.cachedOutputTokens }
  get elapsedMs(): number { return this.startTime > 0 ? Date.now() - this.startTime : 0 }
  get tokensPerSecond(): number {
    if (this.elapsedMs === 0) return 0
    return (this.cachedOutputTokens / this.elapsedMs) * 1000
  }
  
  // Estimation helpers
  estimateRemainingTokens(targetOutputTokens: number): number {
    return Math.max(0, targetOutputTokens - this.cachedOutputTokens)
  }
  
  estimateRemainingTimeMs(targetOutputTokens: number): number {
    if (this.tokensPerSecond === 0) return 0
    return Math.round((this.estimateRemainingTokens(targetOutputTokens) / this.tokensPerSecond) * 1000)
  }
}
```

**Word Boundary 稳定性设计**:
- 避免在任意 chunk 边界计数，防止不稳定
- 只在单词边界（空格后）或超过 50 字符时计数
- Finalize 时强制处理剩余内容

export class IncrementalTokenCounter {
  private lastMessageCount = 0
  private lastTokenCount = 0
  private lastFullHash = ''
  private lastPrefixHash = ''
  private config: Required<IncrementalCounterConfig>
  
  constructor(config: IncrementalCounterConfig = {}) {
    this.config = {
      tokenBudget: config.tokenBudget ?? 100000,
      autoInvalidate: config.autoInvalidate ?? true,
      estimationMultiplier: config.estimationMultiplier ?? 1,
    }
  }
  
  getCount(messages: readonly Message[]): number {
    const hash = getMessageHash(messages)
    
    // Cache hit: same count AND content match
    if (messages.length === this.lastMessageCount && hash === this.lastFullHash) {
      this.stats.hits++
      return this.lastTokenCount
    }
    
    // Cache miss: check if incremental safe
    const isIncrementalSafe =
      messages.length > this.lastMessageCount &&
      this.config.autoInvalidate &&
      this.lastMessageCount > 0
    
    if (isIncrementalSafe) {
      const currentPrefixHash = getMessageHash(messages.slice(0, this.lastMessageCount))
      
      if (currentPrefixHash === this.lastPrefixHash) {
        // Prefix unchanged: add new messages delta
        const newMessages = messages.slice(this.lastMessageCount)
        this.lastTokenCount += roughTokenCountEstimationForMessages(newMessages)
      } else {
        // Prefix changed: full recount
        this.lastTokenCount = roughTokenCountEstimationForMessages(messages)
      }
    } else {
      // Full recount
      this.lastTokenCount = roughTokenCountEstimationForMessages(messages)
    }
    
    this.lastMessageCount = messages.length
    this.lastFullHash = hash
    this.lastPrefixHash = getMessageHash(messages)
    
    return this.lastTokenCount
  }
  
  // Factory presets
  static realtime(): IncrementalTokenCounter {
    return new IncrementalTokenCounter({
      tokenBudget: 50000,
      autoInvalidate: true,
      estimationMultiplier: 1.1,
    })
  }
  
  static batch(): IncrementalTokenCounter {
    return new IncrementalTokenCounter({
      tokenBudget: 200000,
      autoInvalidate: false,
      estimationMultiplier: 1.0,
    })
  }
  
  static lightweight(): IncrementalTokenCounter {
    return new IncrementalTokenCounter({
      tokenBudget: 10000,

### A. ThinkingConfig 详细设计

```typescript
// OpenHorse thinking.ts 设计
export type ThinkingConfig =
  | { type: 'adaptive' }
  | { type: 'enabled'; budgetTokens: number }
  | { type: 'disabled' }

export const THINKING_CONSTANTS = {
  DEFAULT_BUDGET: 10000,
  MAX_BUDGET: 100000,
  MIN_BUDGET: 1024,
  OPUS_BUDGET: 32768,
  SONNET_BUDGET: 10000,
}

export function getThinkingConfig(
  model: string,
  effort?: EffortValue,
): ThinkingConfig {
  // 简化版: 基于 model + effort 决定
  if (!modelSupportsThinking(model)) {
    return { type: 'disabled' }
  }
  
  if (effort === 'high' || effort === 'max') {
    return { type: 'enabled', budgetTokens: THINKING_CONSTANTS.OPUS_BUDGET }
  }
  
  return { type: 'adaptive' }
}
```

### B. Conversation Recovery 简化版

```typescript
// OpenHorse conversationRecovery.ts 设计
export async function loadConversationForResume(
  sessionId: string,
): Promise<ResumeResult> {
  const entries = await loadSessionLog(sessionId)
  
  // Bytes 限制检查
  const totalBytes = estimateBytes(entries)
  if (totalBytes > MAX_RESUME_MESSAGE_BYTES) {
    throw new ResumeError('Session too large', { bytes: totalBytes })
  }
  
  // 简化版: 仅过滤 tool_result unresolved
  const messages = entries.map(entry => {
    if (entry.type === 'user') {
      return filterUnresolvedToolResults(entry)
    }
    return entry
  })
  
  return { messages, sessionId }
}
```

### C. QueryEngine 简化版

```typescript
// OpenHorse QueryEngine.ts 设计
export class QueryEngine {
  private messages: Message[] = []
  private abortController = new AbortController()
  
  async *submitMessage(prompt: string): AsyncGenerator<SDKMessage> {
    // 1. Process user input
    const userInput = await processUserInput(prompt)
    this.messages.push(...userInput.messages)
    
    // 2. Yield system init
    yield buildSystemInit()
    
    // 3. Query loop
    for await (const msg of query(this.messages)) {
      this.messages.push(msg)
      yield normalize(msg)
      
      // Budget check
      if (this.checkBudgetExceeded()) {
        yield errorResult('budget_exceeded')
        return
      }
    }
    
    // 4. Final result
    yield successResult(this.messages)
  }
  
  interrupt(): void {
    this.abortController.abort()
  }
}
```
