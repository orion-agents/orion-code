```typescript
// src/agents/loader.ts
export async function loadAgentsFromDir(dir: string): Promise<AgentDefinition[]> {
  const agentFiles = await glob(`${dir}/*.md`);
  const agents: AgentDefinition[] = [];

  for (const file of agentFiles) {
    const content = await readFile(file);
    const { frontmatter, body } = parseMarkdownFrontmatter(content);

    const agent = parseAgentFromMarkdown(file, frontmatter, body);
    if (agent) agents.push(agent);
  }

  return agents;
}

function parseAgentFromMarkdown(
  filePath: string,
  frontmatter: Record<string, unknown>,
  body: string
): CustomAgentDefinition | null {
  // 必需字段
  const agentType = frontmatter['name'] as string;
  const whenToUse = frontmatter['description'] as string;
  if (!agentType || !whenToUse) return null;

  // 可选字段
  const tools = parseToolsFromFrontmatter(frontmatter['tools']);
  const disallowedTools = parseToolsFromFrontmatter(frontmatter['disallowedTools']);
  const model = frontmatter['model'] as string | undefined;
  const permissionMode = frontmatter['permissionMode'] as PermissionMode | undefined;
  const maxTurns = parseInt(frontmatter['maxTurns'] as string) || undefined;
  const background = frontmatter['background'] === 'true';
  const initialPrompt = frontmatter['initialPrompt'] as string | undefined;

  return {
    agentType,
    whenToUse: whenToUse.replace(/\\n/g, '\n'),
    ...(tools ? { tools } : {}),
    ...(disallowedTools ? { disallowedTools } : {}),
    ...(model ? { model } : {}),
    ...(permissionMode ? { permissionMode } : {}),
    ...(maxTurns ? { maxTurns } : {}),
    ...(background ? { background } : {}),
    ...(initialPrompt ? { initialPrompt } : {}),
    getSystemPrompt: () => body.trim(),
    source: 'user',
    filename: basename(filePath, '.md'),
  };
}
```

### 4. runAgent 执行器

**目标**: 实现 Agent 执行循环

**OpenClaude 参考**: `src/tools/AgentTool/runAgent.ts`

**技术方案**:

```typescript
// src/agents/runner.ts
export async function runAgent(
  agent: AgentDefinition,
  prompt: string,
  parentContext: AgentContext
): Promise<AgentResult> {
  // 1. 构建系统提示词
  const systemPrompt = agent.getSystemPrompt(parentContext);

  // 2. 解析工具集
  const tools = resolveAgentTools(agent, parentContext.tools);

  // 3. 构建初始消息
  const messages: Message[] = [
    { role: 'system', content: systemPrompt },
    ...(agent.initialPrompt ? [{ role: 'user', content: agent.initialPrompt }] : []),
    { role: 'user', content: prompt },
  ];

  // 4. 创建 Agent 上下文
  const agentContext = createSubagentContext(parentContext, {
    agentType: agent.agentType,
    tools,
    permissionMode: agent.permissionMode,
    maxTurns: agent.maxTurns ?? 20,
  });

  // 5. 执行查询循环
  let turn = 0;
  const maxTurns = agent.maxTurns ?? 20;

  while (turn < maxTurns) {
    turn++;

    const response = await parentContext.llm.chatWithTools(
      messages,
      tools,
      agentContext.toolExecutor,
      {
        model: resolveAgentModel(agent, parentContext),
        onChunk: (chunk) => agentContext.onProgress?.(chunk),
      }
    );

    // 添加助手消息
    messages.push({ role: 'assistant', content: response.content });

    // 无工具调用 → 完成
    if (!response.toolCalls || response.toolCalls.length === 0) {
      return {
        summary: response.content,
        usage: response.usage,
        turns: turn,
      };
    }

    // 执行工具 → 继续下一轮
    for (const tc of response.toolCalls) {
      const result = await agentContext.toolExecutor(tc.function.name, tc.function.arguments);
      messages.push({ role: 'tool', content: result, tool_call_id: tc.id });
    }
  }

  return {
    summary: 'Max turns reached',
    usage: { inputTokens: 0, outputTokens: 0 },
    turns: turn,
    maxTurnsReached: true,
  };
}

function resolveAgentTools(
  agent: AgentDefinition,
  parentTools: Tool[]
): Tool[] {
  if (!agent.tools) return parentTools;

  // 白名单过滤
  const allowedSet = new Set(agent.tools);
  let tools = parentTools.filter(t => allowedSet.has(t.name));

  // 黑名单过滤
  if (agent.disallowedTools) {
    const disallowedSet = new Set(agent.disallowedTools);
    tools = tools.filter(t => !disallowedSet.has(t.name));
  }

  return tools;
}
```

```typescript
// src/security/classifier.ts
const DANGEROUS_PATTERNS = [
  // Code from External
  { pattern: /curl\s+.*\|\s*(bash|sh)/i, category: 'code_from_external' },
  { pattern: /wget\s+.*\|\s*(bash|sh)/i, category: 'code_from_external' },
  { pattern: /iex\s*\(\s*iwr/i, category: 'code_from_external' },  // PowerShell
  
  // Irreversible Destruction
  { pattern: /rm\s+(-rf|--recursive|--force)/i, category: 'irreversible_destruction' },
  { pattern: /Remove-Item\s+.*-Recurse.*-Force/i, category: 'irreversible_destruction' },
  { pattern: />\s*\S+/i, category: 'irreversible_destruction' },  // > file
  
  // Credential Exposure
  { pattern: /echo\s+\$[A-Z_]+/i, category: 'credential_exposure' },
  { pattern: /cat\s+.*\.ssh/i, category: 'credential_exposure' },
  { pattern: /cat\s+.*\.env/i, category: 'credential_exposure' },
  
  // Unauthorized Persistence
  { pattern: /\.bashrc/i, category: 'unauthorized_persistence' },
  { pattern: /crontab\s+-e/i, category: 'unauthorized_persistence' },
  { pattern: /\$PROFILE/i, category: 'unauthorized_persistence' },  // PowerShell
  
  // Security Weaken
  { pattern: /chmod\s+777/i, category: 'security_weaken' },
  { pattern: /setenforce\s+0/i, category: 'security_weaken' },
];

export interface SecurityCheckResult {
  safe: boolean;
  category?: string;
  pattern?: string;
  reason?: string;
}

export function checkSecurity(toolName: string, input: Record<string, unknown>): SecurityCheckResult {
  // 1. 工具级安全标记
  const toolClassifierInput = input._tool?.toAutoClassifierInput?.(input) ?? '';
  if (toolClassifierInput === '') return { safe: true };  // 工具声明无安全相关

  // 2. 模式匹配检查
  const inputStr = typeof toolClassifierInput === 'string'
    ? toolClassifierInput
    : JSON.stringify(toolClassifierInput);

  for (const { pattern, category } of DANGEROUS_PATTERNS) {
    if (pattern.test(inputStr)) {
      return {
        safe: false,
        category,
        pattern: pattern.source,
        reason: `Matches dangerous pattern for ${category}`,
      };
    }
  }

  return { safe: true };
}
```
