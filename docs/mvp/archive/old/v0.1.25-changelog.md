```typescript
// client.ts
async function connectToServer(
  name: string,
  config: ScopedMcpServerConfig
): Promise<MCPServerConnection> {
  try {
    // 1. 创建 Transport
    const transport = await createTransport(config)
    
    // 2. 创建 Client
    const client = new Client({ name, version: '1.0.0' }, {
      capabilities: { roots: { listChanged: true } }
    })
    
    // 3. 连接
    await client.connect(transport)
    
    // 4. 获取 Server Info
    const serverInfo = await client.request(
      { method: 'server/info' },
      ServerInfoSchema
    )
    
    // 5. 获取 Instructions
    const instructions = serverInfo.instructions
    
    // 6. 获取 Capabilities
    const capabilities = client.getServerCapabilities()
    
    return {
      client,
      name,
      type: 'connected',
      capabilities,
      serverInfo,
      instructions,
      config,
      cleanup: async () => {
        await client.close()
        await transport.close()
      }
    }
    
  } catch (error) {
    // 401 → needs-auth
    if (error instanceof McpAuthError) {
      return { name, type: 'needs-auth', config }
    }
    
    // 其他 → failed
    return {
      name,
      type: 'failed',
      config,
      error: error.message
    }
  }
}

async function createTransport(config: McpServerConfig): Promise<Transport> {
  switch (config.type) {
    case 'stdio':
      return new StdioClientTransport({
        command: config.command,
        args: config.args,
        env: config.env
      })
    case 'sse':
      return new SSEClientTransport({
        url: new URL(config.url),
        headers: config.headers,
        oauth: config.oauth
      })
    case 'http':
      return new StreamableHTTPClientTransport({
        url: new URL(config.url),
        headers: config.headers
      })
    case 'ws':
      return new WebSocketTransport({
        url: new URL(config.url),
        headers: config.headers
      })

  }
```

```typescript
  scope: AgentMemoryScope
```

```typescript
  const localMemoryPath = getMemoryDirForScope('local')

  }
  
  return { action: 'none' }
}
```

```typescript
  toolName: string           // 工具名
  ruleContent?: string       // 规则内容 (如 "git *")

type PermissionBehavior = 'allow' | 'deny' | 'ask'

type PermissionRule = {

  ruleBehavior: PermissionBehavior
  ruleValue: PermissionRuleValue

// 示例规则
// Bash(git *): allow  → { toolName: 'Bash', ruleContent: 'git *', source: 'userSettings', ruleBehavior: 'allow' }
// Read(.claude/*): deny → { toolName: 'Read', ruleContent: '.claude/*', source: 'projectSettings', ruleBehavior: 'deny' }
```

**Permission Rule Parser**:
```typescript
// permissionRuleParser.ts
function permissionRuleValueFromString(rule: string): PermissionRuleValue {
  // 格式: ToolName 或 ToolName(content)
  // 例: Bash → { toolName: 'Bash' }
  // 例: Bash(git *) → { toolName: 'Bash', ruleContent: 'git *' }
  
  const match = rule.match(/^([^(]+)(?:\(([^)]+)\))?$/)
  if (!match) throw new Error(`Invalid rule: ${rule}`)
  
  return {
    toolName: match[1],
    ruleContent: match[2]
  }
}

function permissionRuleValueToString(value: PermissionRuleValue): string {
  if (value.ruleContent) {
    return `${value.toolName}(${value.ruleContent})`
  }
  return value.toolName
}
```

```typescript
function createPermissionRequestMessage(
  toolName: string,
  decisionReason?: PermissionDecisionReason
): string {
  if (decisionReason) {
    switch (decisionReason.type) {
      case 'classifier':
        return `Classifier '${decisionReason.classifier}' requires approval for this ${toolName} command: ${decisionReason.reason}`
      case 'hook':
        return decisionReason.reason
          ? `Hook '${decisionReason.hookName}' blocked this action: ${decisionReason.reason}`
          : `Hook '${decisionReason.hookName}' requires approval for this ${toolName} command`
      case 'rule':
        const ruleString = permissionRuleValueToString(decisionReason.rule.ruleValue)
        const sourceString = permissionRuleSourceDisplayString(decisionReason.rule.source)
        return `Permission rule '${ruleString}' from ${sourceString} requires approval for this ${toolName} command`
      case 'subcommandResults':
        // 多命令情况
        const needsApproval: string[] = []
        for (const [cmd, result] of decisionReason.reasons) {
          if (result.behavior === 'ask' || result.behavior === 'passthrough') {
            needsApproval.push(cmd)
          }
        }
        return `This ${toolName} command contains multiple operations. The following parts require approval: ${needsApproval.join(', ')}`
      // ...
    }
  }
  return `Permission required for ${toolName} command`
}
```

#### 4. Denial Tracking 防滥用机制

**核心文件**: `src/utils/permissions/denialTracking.ts` (46 行)

**Denial Tracking State**:
```typescript
type DenialTrackingState = {
  consecutiveDenials: number   // 连续拒绝次数
  totalDenials: number         // 总拒绝次数
}

const DENIAL_LIMITS = {
  maxConsecutive: 3,           // 连续拒绝上限
  maxTotal: 20,                // 总拒绝上限
} as const

function createDenialTrackingState(): DenialTrackingState {
  return {
    consecutiveDenials: 0,
    totalDenials: 0,
  }
}

function recordDenial(state: DenialTrackingState): DenialTrackingState {
  return {
    ...state,
    consecutiveDenials: state.consecutiveDenials + 1,
    totalDenials: state.totalDenials + 1,
  }
}

function recordSuccess(state: DenialTrackingState): DenialTrackingState {
  if (state.consecutiveDenials === 0) return state  // No change needed
  return {
    ...state,
    consecutiveDenials: 0,  // 重置连续拒绝
  }
}

function shouldFallbackToPrompting(state: DenialTrackingState): boolean {
  return (
    state.consecutiveDenials >= DENIAL_LIMITS.maxConsecutive ||
    state.totalDenials >= DENIAL_LIMITS.maxTotal
  )
}
```

**用途**:
- 在 auto 模式下，如果连续拒绝 3 次，fallback 到提示用户
- 如果总拒绝次数超过 20 次，也 fallback
- 成功后重置 consecutiveDenials（但保留 totalDenials）

```typescript
export async function connectToServer(
  name: string,

  try {
    // 1. 创建 Transport

      return { status: 'connected', client, name, cleanup: async () => client.kill() }
    }
    
    if (config.type === 'sse' || config.type === 'http') {
      // SSE/HTTP 使用 fetch-based protocol
      const client = await connectHttp(config)
      return { status: 'connected', client, name, cleanup: async () => {} }
    }
    
    return { status: 'failed', name, error: 'Unknown transport type' }
    
  } catch (error) {
    return { status: 'failed', name, error: error.message }
  }
}

async function createTransport(config: McpConfig): Promise<any> {
  switch (config.type) {
    case 'stdio':
      return spawn(config.command, config.args || [], { env: config.env })
    case 'sse':
      return new SSEClient(new URL(config.url), config.headers)
    case 'http':
      return new HttpClient(new URL(config.url), config.headers)
    case 'ws':
      return new WebSocketClient(new URL(config.url), config.headers)
  }
}
```

```typescript
// src/permissions/parser.ts
export function parsePermissionRule(rule: string): PermissionRuleValue {
  // 格式: ToolName 或 ToolName(content)
  // 例: Bash → { toolName: 'Bash' }
  // 例: Bash(git *) → { toolName: 'Bash', ruleContent: 'git *' }
  // 例: Read(.claude/*) → { toolName: 'Read', ruleContent: '.claude/*' }
  
  const match = rule.match(/^([^(]+)(?:\(([^)]+)\))?$/)
  if (!match) {
    throw new Error(`Invalid permission rule format: ${rule}`)
  }
  
  return {
    toolName: match[1].trim(),
    ruleContent: match[2]?.trim()
  }
}

export function formatPermissionRule(value: PermissionRuleValue): string {
  if (value.ruleContent) {
    return `${value.toolName}(${value.ruleContent})`
  }
  return value.toolName
}
```

### 8. Permission Rules 匹配

**目标**: 实现规则匹配逻辑

**OpenClaude 参考**: `src/utils/permissions/permissions.ts`

**技术方案**:

```typescript
// src/permissions/matcher.ts
export function checkPermissionRules(
  toolName: string,
  input: Record<string, unknown>,
  rules: PermissionRulesConfig
): PermissionResult {
  // 优先级: deny > allow > ask
  
  // 1. 检查 deny 规则
  for (const source of ['session', 'projectSettings', 'userSettings']) {
    const denyRules = rules[source]?.deny || []
    for (const rule of denyRules) {
      if (matchesRule(rule, toolName, input)) {
        return { behavior: 'deny', reason: `Denied by rule: ${rule} (${source})` }
      }
    }
  }
  
  // 2. 检查 allow 规则
  for (const source of ['session', 'projectSettings', 'userSettings']) {
    const allowRules = rules[source]?.allow || []
    for (const rule of allowRules) {
      if (matchesRule(rule, toolName, input)) {
        return { behavior: 'allow' }
      }
    }
  }
  
  // 3. 检查 ask 规则
  for (const source of ['session', 'projectSettings', 'userSettings']) {
    const askRules = rules[source]?.ask || []
    for (const rule of askRules) {
      if (matchesRule(rule, toolName, input)) {
        return { behavior: 'ask', reason: `Ask required by rule: ${rule} (${source})` }
      }
    }
  }
  
  // 4. 默认: ask
  return { behavior: 'ask' }
}

function matchesRule(rule: string, toolName: string, input: Record<string, unknown>): boolean {
  const value = parsePermissionRule(rule)
  
  // 工具名匹配
  if (value.toolName !== toolName) return false
  
  // 规则内容匹配
  if (value.ruleContent) {
    return matchRuleContent(value.ruleContent, toolName, input)
  }
  
  return true
}

function matchRuleContent(content: string, toolName: string, input: Record<string, unknown>): boolean {
  // Bash: 匹配命令
  if (toolName === 'Bash' && input.command) {
    return matchPattern(content, input.command as string)
  }
  
  // Read/Edit/Write: 匹配路径
  if (['Read', 'Edit', 'Write'].includes(toolName) && input.file_path) {
    return matchPattern(content, input.file_path as string)
  }
  
  return false
}

function matchPattern(pattern: string, text: string): boolean {
  // * 通配符
  const regex = new RegExp('^' + pattern.replace('*', '.*') + '$')
  return regex.test(text)
}
```

```typescript
export type DenialTrackingState = {
  consecutiveDenials: number
  totalDenials: number
}

export const DENIAL_LIMITS = {
  maxConsecutive: 3,
  maxTotal: 20,
} as const

export function createDenialTrackingState(): DenialTrackingState {
  return { consecutiveDenials: 0, totalDenials: 0 }
}

export function recordDenial(state: DenialTrackingState): DenialTrackingState {
  return {
    ...state,
    consecutiveDenials: state.consecutiveDenials + 1,
    totalDenials: state.totalDenials + 1,
  }
}

export function recordSuccess(state: DenialTrackingState): DenialTrackingState {
  if (state.consecutiveDenials === 0) return state
  return { ...state, consecutiveDenials: 0 }
}

export function shouldFallbackToPrompting(state: DenialTrackingState): boolean {
  return (
    state.consecutiveDenials >= DENIAL_LIMITS.maxConsecutive ||
    state.totalDenials >= DENIAL_LIMITS.maxTotal
  )
}
```
