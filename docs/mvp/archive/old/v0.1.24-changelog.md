**runForkedAgent 执行流程**:
```
runForkedAgent(params)
  ↓
  1. createSubagentContext(toolUseContext, overrides)
     - Clone readFileState
     - New abortController (linked to parent)
     - Wrap getAppState (shouldAvoidPermissionPrompts=true)
     - No-op setAppState/setResponseLength (isolated)
     - Fresh: nestedMemoryAttachmentTriggers, toolDecisions
  ↓
  2. Build initialMessages = forkContextMessages + promptMessages
  ↓
  3. Record sidechain transcript (optional)
  ↓
  4. for await (message of query({...})) {
       - Accumulate usage from message_delta events
       - Record transcript (assistant/user/progress)
       - onMessage callback
     }
  ↓
  5. Log analytics: tengu_fork_agent_query
     - forkLabel, querySource, durationMs, messageCount
     - Full NonNullableUsage: input/output/cacheRead/cacheCreation
     - cacheHitRate = cacheRead / totalInputTokens
  ↓
  6. Return { messages, totalUsage }
  ↓
  7. finally: clear readFileState, clear initialMessages
```

```typescript
export function routeModel(input: RoutingInput, config: SmartRoutingConfig): RoutingDecision {
  // 1. disabled → strong
  if (!config.enabled) return { model: config.strongModel, complexity: 'strong', reason: 'smart-routing disabled' }
  
  // 2. missing models → strong
  if (!config.simpleModel || !config.strongModel) return { model: config.strongModel, ... }
  
  // 3. same model → strong
  if (config.simpleModel === config.strongModel) return { model: config.strongModel, ... }
  
  // 4. empty input → simple
  if (!trimmed) return { model: config.simpleModel, complexity: 'simple', reason: 'empty user text' }
  
  // 5. first turn → strong (task setup)
  if (input.turnNumber === 1) return { model: config.strongModel, complexity: 'strong', reason: 'first turn of session' }
  
  // 6. has code → strong
  if (hasCode(trimmed)) return { model: config.strongModel, reason: 'contains code block' }
  
  // 7. has strong keyword → strong
  if (hasStrongKeyword(trimmed)) return { model: config.strongModel, reason: 'contains reasoning/planning keyword' }
  
  // 8. multi-paragraph → strong
  if (hasMultiParagraph(trimmed)) return { model: config.strongModel, reason: 'multi-paragraph input' }
  
  // 9. char/word 超限 → strong
  if (trimmed.length > maxChars || countWords(trimmed) > maxWords) return { model: config.strongModel, ... }
  
  // 10. short & simple → simple
  return { model: config.simpleModel, complexity: 'simple', reason: `short (${trimmed.length} chars, ${countWords(trimmed)} words)` }
}
```

```typescript
export function getAutoCompactThreshold(model: string): number {
  const effectiveContextWindow = getEffectiveContextWindowSize(model)
  return effectiveContextWindow - AUTOCOMPACT_BUFFER_TOKENS
}

export function getEffectiveContextWindowSize(model: string): number {
  const reservedTokensForSummary = Math.min(
    getMaxOutputTokensForModel(model),
    MAX_OUTPUT_TOKENS_FOR_SUMMARY,
  )
  let contextWindow = getContextWindowForModel(model, getSdkBetas())
  
  // env override
  const autoCompactWindow = process.env.CLAUDE_CODE_AUTO_COMPACT_WINDOW
  if (autoCompactWindow) {
    const parsed = parseInt(autoCompactWindow, 10)
    if (!isNaN(parsed) && parsed > 0) {
      contextWindow = Math.min(contextWindow, parsed)
    }
  }
  
  // Floor: effective context >= summary + buffer
  return Math.max(contextWindow - reservedTokensForSummary, reservedTokensForSummary + AUTOCOMPACT_BUFFER_TOKENS)
}
```

**shouldAutoCompact 流程**:
```typescript
export async function shouldAutoCompact(
  messages: Message[],
  model: string,
  querySource?: QuerySource,
  snipTokensFreed = 0,
): Promise<boolean> {
  // 1. Recursion guards: session_memory / compact / marble_origami 死锁
  if (querySource === 'session_memory' || querySource === 'compact') return false
  
  // 2. Feature gates: REACTIVE_COMPACT / CONTEXT_COLLAPSE 模式
  if (feature('REACTIVE_COMPACT') && getFeatureValue('tengu_cobalt_raccoon', false)) return false
  if (feature('CONTEXT_COLLAPSE') && isContextCollapseEnabled()) return false
  
  // 3. Token check
  const tokenCount = tokenCountWithEstimation(messages) - snipTokensFreed
  const threshold = getAutoCompactThreshold(model)
  return tokenCount >= threshold
}
```

**autoCompactIfNeeded 流程**:
```typescript
export async function autoCompactIfNeeded(
  messages: Message[],
  toolUseContext: ToolUseContext,
  cacheSafeParams: CacheSafeParams,
  querySource?: QuerySource,
  tracking?: AutoCompactTrackingState,
  snipTokensFreed?: number,
): Promise<{
  wasCompacted: boolean
  compactionResult?: CompactionResult
  consecutiveFailures?: number
}> {
  // 1. Circuit breaker check
  if (tracking?.consecutiveFailures >= MAX_CONSECUTIVE_AUTOCOMPACT_FAILURES) {
    return { wasCompacted: false }
  }
  
  // 2. shouldAutoCompact check
  if (!await shouldAutoCompact(messages, model, querySource, snipTokensFreed)) {
    return { wasCompacted: false }
  }
  
  // 3. Context partitioning + pruning
  const partitioned = partitionContext(messages, { contextWindow, recentCount: 5 })
  if (!partitioned.canFitInWindow) {
    const pruned = pruneByRelevance(nonSystemMessages, {
      targetTokens: availableSpace,
      preserveRecent: 3,
      preserveTools: true,
      preserveErrors: true,
    })
    messages = [...systemMessages, ...pruned]
  }
  
  // 4. Try session memory compact first (experimental)
  const sessionMemoryResult = await trySessionMemoryCompaction(messages, agentId, threshold)
  if (sessionMemoryResult) {
    runPostCompactCleanup(querySource)
    notifyCompaction(querySource, agentId)
    return { wasCompacted: true, compactionResult: sessionMemoryResult }
  }
  
  // 5. Legacy compactConversation
  try {
    const compactionResult = await compactConversation(messages, toolUseContext, cacheSafeParams, true, undefined, true, recompactionInfo)
    runPostCompactCleanup(querySource)
    return { wasCompacted: true, compactionResult, consecutiveFailures: 0 }
  } catch (error) {
    // Increment failure count
    const nextFailures = (tracking?.consecutiveFailures ?? 0) + 1
    if (nextFailures >= MAX_CONSECUTIVE_AUTOCOMPACT_FAILURES) {
      logForDebugging('autocompact: circuit breaker tripped', { level: 'warn' })
    }
    return { wasCompacted: false, consecutiveFailures: nextFailures }
  }
}
```

**calculateTokenWarningState**:
```typescript
export function calculateTokenWarningState(tokenUsage: number, model: string): {
  percentLeft: number
  isAboveWarningThreshold: boolean
  isAboveErrorThreshold: boolean
  isAboveAutoCompactThreshold: boolean
  isAtBlockingLimit: boolean
} {
  const autoCompactThreshold = getAutoCompactThreshold(model)
  const threshold = isAutoCompactEnabled() ? autoCompactThreshold : getEffectiveContextWindowSize(model)
  const rawContextWindow = getContextWindowForModel(model, getSdkBetas())
  
  const percentLeft = Math.max(0, Math.round(((rawContextWindow - tokenUsage) / rawContextWindow) * 100))
  const warningThreshold = threshold - WARNING_THRESHOLD_BUFFER_TOKENS
  const errorThreshold = threshold - ERROR_THRESHOLD_BUFFER_TOKENS
  const blockingLimit = getEffectiveContextWindowSize(model) - MANUAL_COMPACT_BUFFER_TOKENS
  
  return {
    percentLeft,
    isAboveWarningThreshold: tokenUsage >= warningThreshold,
    isAboveErrorThreshold: tokenUsage >= errorThreshold,
    isAboveAutoCompactThreshold: isAutoCompactEnabled() && tokenUsage >= autoCompactThreshold,
    isAtBlockingLimit: tokenUsage >= blockingLimit,
  }
}
```

```typescript
// src/utils/subagentContext.ts
import type { ToolContext, PermissionResult } from '../framework/tool';
import { v4 as uuidv4 } from 'uuid';

export type SubagentContextOverrides = {
  options?: Partial<ToolContext['config']>;
  agentId?: string;
  agentType?: string;
  messages?: Message[];
  readFileState?: Map<string, any>;
  abortController?: AbortController;
  getAppState?: () => any;
  
  // Explicit opt-in to share
  shareSetAppState?: boolean;
  shareSetResponseLength?: boolean;
  shareAbortController?: boolean;
}

export function createSubagentContext(
  parentContext: ToolContext,
  overrides?: SubagentContextOverrides,
): ToolContext {
  // AbortController: 新建，linked to parent
  const abortController = overrides?.abortController ??
    (overrides?.shareAbortController ? parentContext.abortController
     : createLinkedAbortController(parentContext.abortController));
  
  // AppState: wrapped to suppress prompts
  const getAppState: () => any = overrides?.getAppState ??
    (overrides?.shareAbortController ? parentContext.getAppState
     : () => ({
         ...parentContext.getAppState?.() || {},
         shouldAvoidPermissionPrompts: true,
       }));
  
  return {
    // Mutable state: clone/fresh
    readFileState: overrides?.readFileState ? 
      new Map(overrides.readFileState) : new Map(parentContext.readFileState || new Map()),
    toolDecisions: undefined,
    
    // AbortController
    abortController,
    
    // AppState
    getAppState,
    setAppState: overrides?.shareSetAppState ? parentContext.setAppState : () => {},
    setResponseLength: overrides?.shareSetResponseLength ? parentContext.setResponseLength : () => {},
    
    // Overridable
    cwd: overrides?.options?.cwd ?? parentContext.cwd,
    config: overrides?.options ? { ...parentContext.config, ...overrides.options } : parentContext.config,
    
    // New ID
    agentId: overrides?.agentId ?? uuidv4(),
    agentType: overrides?.agentType,
    
    // Query tracking
    queryTracking: {
      chainId: uuidv4(),
      depth: (parentContext.queryTracking?.depth ?? -1) + 1,
    },
  };
}

function createLinkedAbortController(parent: AbortController): AbortController {
  const child = new AbortController();
  parent.signal.addEventListener('abort', () => child.abort());
  return child;
}
```

**技术方案**:

```typescript
// src/utils/runForkedAgent.ts
import type { LLMService, Message, StreamCallbacks, Tool } from '../services/llm';
import type { CacheSafeParams } from './cacheSafeParams';
import type { SubagentContextOverrides } from './subagentContext';
import { query, type QueryEvent } from '../framework/query';
import { createSubagentContext } from './subagentContext';
import { logEvent } from '../services/analytics';

export type ForkedAgentParams = {
  promptMessages: Message[];
  cacheSafeParams: CacheSafeParams;
  canUseTool?: (name: string, args: any) => PermissionResult;
  querySource: string;
  forkLabel: string;
  overrides?: SubagentContextOverrides;
  maxTurns?: number;
  onMessage?: (message: Message) => void;
  skipCacheWrite?: boolean;
}

export type ForkedAgentResult = {
  messages: Message[];
  totalUsage: {
    input_tokens: number;
    output_tokens: number;
    cache_read_input_tokens: number;
    cache_creation_input_tokens: number;
  };
}

export async function runForkedAgent({
  promptMessages,
  cacheSafeParams,
  canUseTool,
  querySource,
  forkLabel,
  overrides,
  maxTurns,
  onMessage,
  skipCacheWrite,
}: ForkedAgentParams): Promise<ForkedAgentResult> {
  const startTime = Date.now();
  const outputMessages: Message[] = [];
  let totalUsage = { input_tokens: 0, output_tokens: 0, cache_read_input_tokens: 0, cache_creation_input_tokens: 0 };
  
  const { systemPrompt, userContext, systemContext, toolUseContext, forkContextMessages } = cacheSafeParams;
  
  // Create isolated context
  const isolatedContext = createSubagentContext(toolUseContext, overrides);
  
  // Build initial messages
  const initialMessages: Message[] = [...forkContextMessages, ...promptMessages];
  
  // Build full system prompt
  const fullSystemPrompt = [
    systemPrompt,
    Object.entries(userContext).map(([k, v]) => `${k}: ${v}`).join('\n'),
    Object.entries(systemContext).map(([k, v]) => `${k}: ${v}`).join('\n'),
  ].filter(Boolean).join('\n\n');
  
  // Run query loop
  try {
    for await (const event of query({
      messages: [{ role: 'system', content: fullSystemPrompt }, ...initialMessages],
      tools: isolatedContext.config.tools || [],
      toolExecutor: async (name, args) => {
        if (canUseTool) {
          const perm = canUseTool(name, args);
          if (perm.behavior === 'deny') return JSON.stringify({ success: false, error: perm.reason });
        }
        const tool = isolatedContext.config.tools?.find(t => t.name === name);
        if (!tool) return JSON.stringify({ success: false, error: `Tool ${name} not found` });
        const result = await tool.execute(args, isolatedContext);
        return JSON.stringify(result);
      },
      llm: isolatedContext.config.llm,
      maxTurns: maxTurns ?? 10,
      abortSignal: isolatedContext.abortController?.signal,
    })) {
      // Track usage
      if (event.type === 'complete' && event.usage) {
        totalUsage.input_tokens += event.usage.promptTokens;
        totalUsage.output_tokens += event.usage.completionTokens;
      }
      
      // Track messages
      if (event.type === 'message') {
        const msg: Message = { role: event.role, content: event.content };
        outputMessages.push(msg);
        onMessage?.(msg);
      }
    }
  } finally {
    isolatedContext.readFileState?.clear();
  }
  
  const durationMs = Date.now() - startTime;
  
  // Log analytics
  const totalInputTokens = totalUsage.input_tokens + totalUsage.cache_creation_input_tokens + totalUsage.cache_read_input_tokens;
  const cacheHitRate = totalInputTokens > 0 ? totalUsage.cache_read_input_tokens / totalInputTokens : 0;
  
  logEvent('fork_agent_query', {
    forkLabel,
    querySource,
    durationMs,
    messageCount: outputMessages.length,
    inputTokens: totalUsage.input_tokens,
    outputTokens: totalUsage.output_tokens,
    cacheReadInputTokens: totalUsage.cache_read_input_tokens,
    cacheCreationInputTokens: totalUsage.cache_creation_input_tokens,
    cacheHitRate,
    queryDepth: isolatedContext.queryTracking?.depth ?? 0,
  });
  
  return { messages: outputMessages, totalUsage };
}
```

## P1 (重要) — Smart Routing 实现

### 4. routeModel 完整实现

**目标**: 10 种判断的完整路由逻辑

**OpenClaude 参考**: `src/services/api/smartModelRouting.ts`

**技术方案**:

```typescript
// src/services/smartRouting.ts
export type SmartRoutingConfig = {
  enabled: boolean;
  simpleModel: string;
  strongModel: string;
  simpleMaxChars?: number;  // 默认 160
  simpleMaxWords?: number;  // 默认 28
}

export type RoutingDecision = {
  model: string;
  complexity: 'simple' | 'strong';
  reason: string;
}

export type RoutingInput = {
  userText: string;
  recentToolUses?: number;
  turnNumber?: number;
}

const DEFAULT_SIMPLE_MAX_CHARS = 160;
const DEFAULT_SIMPLE_MAX_WORDS = 28;

const STRONG_KEYWORDS = [
  'plan', 'design', 'architect', 'architecture', 'refactor',
  'debug', 'investigate', 'analyze', 'analyse', 'implement',
  'optimize', 'optimise', 'review', 'audit', 'diagnose',
  'root cause', 'root-cause', 'why does', 'why is',
  'how should', 'why did', 'propose', 'trace', 'reproduce',
];

const STRONG_KEYWORD_RE = new RegExp(
  `\\b(?:${STRONG_KEYWORDS.map(k => k.replace(/[-]/g, '[-\\s]')).join('|')})\\b`,
  'i',
);

const CODE_FENCE_RE = /```[\s\S]*?```|`[^`\n]+`/;

function hasCode(text: string): boolean {
  return CODE_FENCE_RE.test(text);
}

function hasStrongKeyword(text: string): boolean {
  return STRONG_KEYWORD_RE.test(text);
}

function hasMultiParagraph(text: string): boolean {
  return /\n\s*\n/.test(text);
}

function countWords(text: string): number {
  return text.trim().split(/\s+/).length;
}

export function routeModel(
  input: RoutingInput,
  config: SmartRoutingConfig,
): RoutingDecision {
  // 1-3. Disabled / missing / same
  if (!config.enabled) return { model: config.strongModel, complexity: 'strong', reason: 'disabled' };
  if (!config.simpleModel || !config.strongModel) return { model: config.strongModel, complexity: 'strong', reason: 'missing config' };
  if (config.simpleModel === config.strongModel) return { model: config.strongModel, complexity: 'strong', reason: 'same model' };
  
  const text = input.userText ?? '';
  const trimmed = text.trim();
  
  // 4. Empty → simple
  if (!trimmed) return { model: config.simpleModel, complexity: 'simple', reason: 'empty' };
  
  // 5. First turn → strong
  if (input.turnNumber === 1) return { model: config.strongModel, complexity: 'strong', reason: 'first turn' };
  
  const maxChars = config.simpleMaxChars ?? DEFAULT_SIMPLE_MAX_CHARS;
  const maxWords = config.simpleMaxWords ?? DEFAULT_SIMPLE_MAX_WORDS;
  
  // 6-8. Code / keyword / multi-paragraph → strong
  if (hasCode(trimmed)) return { model: config.strongModel, complexity: 'strong', reason: 'has code' };
  if (hasStrongKeyword(trimmed)) return { model: config.strongModel, complexity: 'strong', reason: 'has keyword' };
  if (hasMultiParagraph(trimmed)) return { model: config.strongModel, complexity: 'strong', reason: 'multi-paragraph' };
  
  // 9. Char/word limit → strong
  if (trimmed.length > maxChars) return { model: config.strongModel, complexity: 'strong', reason: `>${maxChars} chars` };
  if (countWords(trimmed) > maxWords) return { model: config.strongModel, complexity: 'strong', reason: `>${maxWords} words` };
  
  // 10. Short → simple
  return {
    model: config.simpleModel,
    complexity: 'simple',
    reason: `short (${trimmed.length}c, ${countWords(trimmed)}w)`,
  };
}
```

```typescript
// src/services/compact/threshold.ts
const MAX_OUTPUT_TOKENS_FOR_SUMMARY = 20_000;
const AUTOCOMPACT_BUFFER_TOKENS = 13_000;
const WARNING_THRESHOLD_BUFFER_TOKENS = 20_000;
const ERROR_THRESHOLD_BUFFER_TOKENS = 20_000;
const MANUAL_COMPACT_BUFFER_TOKENS = 3_000;
const MAX_CONSECUTIVE_FAILURES = 3;

export type AutoCompactState = {
  compacted: boolean;
  turnCounter: number;
  turnId: string;
  consecutiveFailures: number;
}

// Model context windows
const MODEL_CONTEXT_WINDOWS: Record<string, number> = {
  'claude-opus-4-7': 200_000,
  'claude-sonnet-4-6': 200_000,
  'claude-haiku-4-5': 200_000,
  'gpt-4o': 128_000,
  'gpt-4o-mini': 128_000,
  'glm-4': 128_000,
};

export function getContextWindowForModel(model: string): number {
  // Try exact match
  if (MODEL_CONTEXT_WINDOWS[model]) return MODEL_CONTEXT_WINDOWS[model];
  
  // Try prefix match
  for (const [key, value] of Object.entries(MODEL_CONTEXT_WINDOWS)) {
    if (model.startsWith(key) || key.startsWith(model)) return value;
  }
  
  // Default
  return 128_000;
}

export function getAutoCompactThreshold(model: string): number {
  const contextWindow = getContextWindowForModel(model);
  return contextWindow - AUTOCOMPACT_BUFFER_TOKENS;
}

export function calculateTokenWarningState(
  tokenUsage: number,
  model: string,
): {
  percentLeft: number;
  isAboveWarning: boolean;
  isAboveError: boolean;
  isAboveAutoCompact: boolean;
  isAtBlocking: boolean;
} {
  const threshold = getAutoCompactThreshold(model);
  const rawContext = getContextWindowForModel(model);
  
  const percentLeft = Math.max(0, Math.round(((rawContext - tokenUsage) / rawContext) * 100));
  const warningThreshold = threshold - WARNING_THRESHOLD_BUFFER_TOKENS;
  const errorThreshold = threshold - ERROR_THRESHOLD_BUFFER_TOKENS;
  const blockingLimit = rawContext - MANUAL_COMPACT_BUFFER_TOKENS;
  
  return {
    percentLeft,
    isAboveWarning: tokenUsage >= warningThreshold,
    isAboveError: tokenUsage >= errorThreshold,
    isAboveAutoCompact: tokenUsage >= threshold,
    isAtBlocking: tokenUsage >= blockingLimit,
  };
}
```

```typescript
import type { Message } from '../llm';

export function estimateMessagesTokens(messages: Message[]): number {
  let total = 0;
  for (const msg of messages) {
    const content = typeof msg.content === 'string' ? msg.content : '';
    // Rough estimate: 1 token ≈ 4 chars
    total += Math.ceil(content.length / 4);
  }
  return total;
}

export function shouldAutoCompact(
  messages: Message[],
  model: string,
  querySource?: string,
  consecutiveFailures?: number,
): boolean {
  // 1. Circuit breaker
  if (consecutiveFailures >= MAX_CONSECUTIVE_FAILURES) return false;
  
  // 2. Recursion guards
  if (querySource === 'session_memory' || querySource === 'compact') return false;
  
  // 3. Token check
  const tokenCount = estimateMessagesTokens(messages);
  const threshold = getAutoCompactThreshold(model);
  
  return tokenCount >= threshold;
}
```

```typescript
// src/utils/contextPartitioning.ts
export type PriorityZone = 'recent' | 'important' | 'background' | 'system';

export type ZoneConfig = {
  name: PriorityZone;
  maxTokens: number;
  retentionPolicy: 'keep_all' | 'prune_oldest' | 'prune_least_important';
  priority: number;
}

const DEFAULT_ZONES: ZoneConfig[] = [
  { name: 'recent', maxTokens: 50_000, retentionPolicy: 'keep_all', priority: 4 },
  { name: 'important', maxTokens: 30_000, retentionPolicy: 'prune_least_important', priority: 3 },
  { name: 'background', maxTokens: 10_000, retentionPolicy: 'prune_oldest', priority: 2 },
  { name: 'system', maxTokens: 8_000, retentionPolicy: 'keep_all', priority: 1 },
];

export function classifyMessage(message: Message, isRecent: boolean): PriorityZone {
  const content = typeof message.content === 'string' ? message.content : '';
  
  if (message.role === 'system') return 'system';
  if (content.includes('error') || content.includes('fail')) return 'important';
  if (content.length > 2000) return 'important';
  if (isRecent) return 'recent';
  return 'background';
}

export function partitionContext(
  messages: Message[],
  options?: { contextWindow?: number; recentCount?: number },
): Map<PriorityZone, Message[]> {
  const recentCount = options?.recentCount ?? 5;
  const zones = new Map<PriorityZone, Message[]>();
  
  for (const zone of DEFAULT_ZONES) {
    zones.set(zone.name, []);
  }
  
  for (let i = 0; i < messages.length; i++) {
    const msg = messages[i];
    const isRecent = i >= messages.length - recentCount;
    const zone = classifyMessage(msg, isRecent);
    zones.get(zone)?.push(msg);
  }
  
  return zones;
}
```
