### 1. Command Types 定义

```typescript
// src/types/command.ts
type Command = PromptCommand | LocalCommand | LocalJSXCommand

type PromptCommand = {
  type: 'prompt'
  name: string
  description: string
  aliases?: string[]
  hasUserSpecifiedDescription?: boolean
  progressMessage: string
  contentLength: number
  argNames?: string[]
  allowedTools?: string[]
  model?: string
  source: SettingSource | 'builtin' | 'mcp' | 'plugin' | 'bundled'
  pluginInfo?: { pluginManifest: PluginManifest; repository: string }
  disableNonInteractive?: boolean
  hooks?: HooksSettings
  skillRoot?: string
  context?: 'inline' | 'fork'  // 执行上下文
  agent?: string               // Fork 时的 Agent 类型
  effort?: EffortValue
  paths?: string[]             // 文件路径匹配
  getPromptForCommand(args: string, context: ToolUseContext): Promise<ContentBlockParam[]>
}

type LocalCommand = {
  type: 'local'
  name: string
  description: string
  aliases?: string[]
  supportsNonInteractive: boolean
  load: () => Promise<LocalCommandModule>
}

type LocalJSXCommand = {
  type: 'local-jsx'
  name: string
  description: string
  aliases?: string[]
  isEnabled?: () => boolean
  isHidden?: boolean
  load: () => Promise<{ default: Command }>
}

type LocalCommandResult =
  | { type: 'text'; value: string }
  | { type: 'compact'; compactionResult: CompactionResult; displayText?: string }
  | { type: 'skip' }
```

### Phase A: 类型安全重构 (#2.4)

**文件**: `src/core/agent.ts`, `src/core/brain.ts`

**当前问题**:

**重构方案**:
```typescript
// 引入鉴别联合 + 泛型
export interface SkillContext<P = Record<string, unknown>> {
  name: string;
  params?: P;
}

export type TaskResultData =
  | { kind: 'file'; content: string }
  | { kind: 'command'; stdout: string; stderr?: string; exitCode: number }
  | { kind: 'metrics'; steps?: number; metrics?: Record<string, number> }
  | { kind: 'generic'; value: unknown };

export interface TaskResult<D extends TaskResultData = TaskResultData> {
  success: boolean;
  data?: D;
  error?: string;
  duration?: number;
}

interface BrainStatus {
  pendingTasks: number;
  activeAgents: string[];
  lastError?: string;
}
```

```typescript
export async function searchMemoriesAsync(
```
