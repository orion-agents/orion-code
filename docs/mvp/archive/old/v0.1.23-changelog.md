### 2. runForkedAgent 实现

**目标**: 实现子 Agent 执行，共享父 Prompt Cache

```typescript
// src/utils/forkedAgent.ts
export type ForkedAgentParams = {
  promptMessages: Message[]
  cacheSafeParams: CacheSafeParams
  canUseTool: CanUseToolFn
  querySource: string
  forkLabel: string
  maxTurns?: number
  onMessage?: (message: Message) => void
}

export type ForkedAgentResult = {
  messages: Message[]
  totalUsage: { input_tokens: number; output_tokens: number }
}

export async function runForkedAgent(params: ForkedAgentParams): Promise<ForkedAgentResult> {
  // 1. 创建子 context
  const subContext = createSubagentContext(params.cacheSafeParams.toolUseContext)

  // 2. 合并消息
  const messages = [...params.cacheSafeParams.forkContextMessages, ...params.promptMessages]

  // 3. 执行 query loop
  const yieldedMessages: Message[] = []
  let totalUsage = { input_tokens: 0, output_tokens: 0 }

  for await (const message of query(subContext)) {
    yieldedMessages.push(message)
    if (message.usage) {
      totalUsage.input_tokens += message.usage.input_tokens
      totalUsage.output_tokens += message.usage.output_tokens
    }
    if (params.maxTurns && turnCount >= params.maxTurns) break
    if (params.onMessage) params.onMessage(message)
  }

  return { messages: yieldedMessages, totalUsage }
}
```

### 4. Smart Routing 实现

**目标**: 智能模型路由

```typescript
// src/services/api/smartRouting.ts
export type SmartRoutingConfig = {
  enabled: boolean
  simpleModel: string
  strongModel: string
  simpleMaxChars?: number  // 默认 160
  simpleMaxWords?: number  // 默认 28
}

const STRONG_KEYWORDS = [
  'plan', 'design', 'architect', 'refactor', 'debug',
  'investigate', 'analyze', 'implement', 'optimize', 'review'
]

export function routeModel(
  input: { userText: string; turnNumber?: number },
  config: SmartRoutingConfig
): { model: string; complexity: 'simple' | 'strong'; reason: string } {
  // 1. 检查是否启用
  if (!config.enabled) return { model: config.strongModel, complexity: 'strong', reason: 'disabled' }

  // 2. First turn 用 strong
  if (input.turnNumber === 1) return { model: config.strongModel, complexity: 'strong', reason: 'first turn' }

  // 3. Strong keywords
  if (STRONG_KEYWORDS.some(kw => input.userText.toLowerCase().includes(kw))) {
    return { model: config.strongModel, complexity: 'strong', reason: 'keyword' }
  }

  // 4. 简短输入用 simple
  const charCount = input.userText.length
  const wordCount = input.userText.split(/\s+/).length
  if (charCount <= 160 && wordCount <= 28) {
    return { model: config.simpleModel, complexity: 'simple', reason: 'short' }
  }

  return { model: config.strongModel, complexity: 'strong', reason: 'default' }
}
```

```typescript
export type ProviderOverride = {
  model: string
  baseURL: string

}

export type AgentRoutingConfig = {
  agentRouting: Record<string, string>  // agentType → modelName
  agentModels: Record<string, { base_url: string; api_key: string }>
}

export function resolveAgentProvider(
  agentType: string,
  config: AgentRoutingConfig | null
): ProviderOverride | null {
  if (!config) return null

  // Priority: agentType > "default" > null
  const modelName = config.agentRouting[agentType] || config.agentRouting['default']
  if (!modelName) return null

  const modelConfig = config.agentModels[modelName]
  if (!modelConfig) return null

  return {
    model: modelName,
    baseURL: modelConfig.base_url,
    apiKey: modelConfig.api_key
  }
}
```

### 7. Relevance Pruning

**目标**: 相关性剪枝

```typescript
// src/utils/relevancePruning.ts
export type PruningOptions = {
  targetTokens: number
  taskContext?: string
  preserveRecent?: number
}

const STOP_WORDS = new Set(['the', 'and', 'for', 'are', 'but', 'not'])

export function extractKeywords(text: string): Set<string> {
  const words = text.toLowerCase().split(/\s+/)
  const keywords = new Set<string>()
  for (const word of words) {
    const cleaned = word.replace(/[^a-z]/g, '')
    if (cleaned.length > 3 && !STOP_WORDS.has(cleaned)) keywords.add(cleaned)
  }
  return keywords
}

export function calculateRelevance(message: Message, taskContext: string): number {
  const keywords1 = extractKeywords(getTextContent(message))
  const keywords2 = extractKeywords(taskContext)
  let overlap = 0
  for (const kw of keywords1) if (keywords2.has(kw)) overlap++
  return (2 * overlap) / (keywords1.size + keywords2.size)
}
```

## OpenClaude vs OpenHorse 对比分析

| OpenClaude 设计 | OpenHorse v0.1.23 实现 | 简化点 |
|----------------|----------------------|-------|
| Fork + Transcript + Analytics | Fork + Analytics 仅 | 去掉 Transcript Recording |
| Side Query + Fingerprint + OAuth | Side Query 仅 | 去掉 Fingerprint/OAuth |
| Auto Compact + Circuit + State | Threshold + Circuit 仅 | 去掉 State tracking |
| Smart Routing + Keyword + Code + Recent | Keyword 仅 | 去掉 Code/Recent 检测 |
| Agent Routing + Normalized + Collision | Direct Lookup 仅 | 去掉 Normalized/Collision |
| Partitioning + Zone + Retention | Zone 分类仅 | 去掉 Retention Policy |
| Relevance + StopWords + Preserve | Keyword Overlap 仅 | 去掉 Preserve Options |

### OpenHorse 差异化优势

1. **Forked Agent 核心** — Prompt Cache 共享，节省成本
2. **Auto Compact 核心** — Buffer + Circuit Breaker
3. **Smart Routing 核心** — Keyword 检测 + Simple/Strong
4. **Agent Routing 简化** — Direct Provider Override
5. **Context Management** — Priority Zone + Keyword Pruning
6. **渐进式架构** — v0.1.22 Hooks → v0.1.23 Fork/Routing/Context
