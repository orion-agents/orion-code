**Microcompact 设计**:

```typescript
// src/services/compact/microCompact.ts
// Microcompact: 修剪不必要内容，不调用模型

type MicrocompactOptions = {
  maxToolResultChars: number      // 工具结果截断阈值
  preserveRecentTools: number     // 保留最近 N 个工具
}

async function microcompactMessages(
  messages: Message[],
  context: ToolUseContext
): Promise<{ messages: Message[]; tokensSaved: number }> {
  const processed: Message[] = []

  for (const msg of messages) {
    if (msg.role === 'tool') {
      // 修剪工具结果
      const truncated = truncateToolResult(msg.content, MAX_TOOL_RESULT_CHARS)
      processed.push({ ...msg, content: truncated })
    } else if (msg.role === 'assistant') {
      // 修剪 assistant 消息中的冗余内容
      const trimmed = trimAssistantMessage(msg)
      processed.push(trimmed)
    } else {
      processed.push(msg)
    }
  }

  return { messages: processed, tokensSaved: estimateTokensSaved(messages, processed) }
}

function truncateToolResult(content: string, maxChars: number): string {
  if (content.length <= maxChars) return content
  return content.slice(0, maxChars) + '\n...[truncated]'
}
```

```typescript
// Transport 接口
interface Transport {
  connect(): Promise<void>
  close(): void
  write(message: StdoutMessage): Promise<void>
  setOnData(callback: (data: string) => void): void
  setOnClose(callback: () => void): void
}

// SSE Transport (Server-Sent Events)
class SSETransport implements Transport {
  private eventSource: EventSource

  async connect(): Promise<void> {
    this.eventSource = new EventSource(this.url, {
      headers: this.headers
    })
    this.eventSource.onmessage = (event) => {
      this.onDataCallback(event.data)
    }
    this.eventSource.onerror = (error) => {
      this.onCloseCallback()
    }
  }

  async write(message: StdoutMessage): Promise<void> {
    // SSE 是单向的，write 需要通过 POST
    await axios.post(this.postUrl, message, {
      headers: this.headers
    })
  }
}

// WebSocket Transport
class WebSocketTransport implements Transport {
  private ws: WebSocket

  async connect(): Promise<void> {
    this.ws = new WebSocket(this.url, {
      headers: this.headers
    })
    this.ws.onmessage = (event) => {
      this.onDataCallback(event.data.toString())
    }
    this.ws.onclose = () => {
      this.onCloseCallback()
    }
  }

  async write(message: StdoutMessage): Promise<void> {
    this.ws.send(JSON.stringify(message))
  }
}

// Hybrid Transport (WebSocket 读 + HTTP POST 写)
class HybridTransport extends WebSocketTransport {
  private postUrl: string
  private uploader: SerialBatchEventUploader

  // 写入通过 HTTP POST (批量、重试、backpressure)
  async write(message: StdoutMessage): Promise<void> {
    if (message.type === 'stream_event') {
      // 延迟批量发送 (减少 POST 数)
      this.streamEventBuffer.push(message)
      if (!this.streamEventTimer) {
        this.streamEventTimer = setTimeout(
          () => this.flushStreamEvents(),
          BATCH_FLUSH_INTERVAL_MS  // 100ms
        )
      }
    } else {
      // 立即发送 (flush buffer first)
      await this.uploader.enqueue([...this.takeStreamEvents(), message])
      await this.uploader.flush()
    }
  }

  // 批量写入
  async writeBatch(messages: StdoutMessage[]): Promise<void> {
    await this.uploader.enqueue([...this.takeStreamEvents(), ...messages])
    await this.uploader.flush()
  }
}
```

### 3. Session Recovery 恢复机制

```typescript
// src/session/recovery.ts
export interface RecoveryOptions {
  compactBoundary: boolean    // 是否只加载 compact 后的消息
  maxMessages?: number        // 最大加载消息数
}

export async function recoverSession(
  sessionId: string,
  options?: RecoveryOptions
): Promise<Message[]> {
  const { messages, metadata } = await loadSessionForResume(sessionId)

  // 应用 compact boundary
  if (options?.compactBoundary && metadata.compactBoundary) {
    return messages.slice(metadata.compactBoundary)
  }

  // 应用最大消息数限制
  if (options?.maxMessages && messages.length > options.maxMessages) {
    // 保留最近的 N 条消息
    return messages.slice(-options.maxMessages)
  }

  return messages
}

export function getMessagesAfterCompactBoundary(
  messages: Message[]
): Message[] {

    m.role === 'system' && m.content.includes('compact_boundary')
  )
  if (boundaryIndex === -1) return messages
  return messages.slice(boundaryIndex + 1)
}
```

```typescript
export type ThinkingConfig =
  | { type: 'disabled' }
  | { type: 'enabled'; budgetTokens: number }
  | { type: 'adaptive' }

export function getThinkingConfig(
  model: string,
  userConfig?: ThinkingConfig
): ThinkingConfig {
  // 1. 检查模型支持
  if (!modelSupportsThinking(model)) {
    return { type: 'disabled' }
  }

  // 2. 用户配置覆盖
  if (userConfig) return userConfig

  // 3. 默认启用 adaptive (支持的模型)
  if (modelSupportsAdaptiveThinking(model)) {
    return { type: 'adaptive' }
  }

  // 4. 其他支持模型默认 disabled (用户可启用)
  return { type: 'disabled' }
}

export function modelSupportsThinking(model: string): boolean {
  const m = model.toLowerCase()
  // Claude 4+
  if (m.includes('claude-4') || m.includes('opus-4') || m.includes('sonnet-4')) {
    return true
  }
  // DeepSeek reasoner
  if (m.includes('deepseek-reasoner') || m.includes('deepseek-r1')) {
    return true
  }
  return false
}

export function modelSupportsAdaptiveThinking(model: string): boolean {
  const m = model.toLowerCase()
  return m.includes('opus-4-7') || m.includes('opus-4-6') || m.includes('sonnet-4-6')
}

// Ultrathink 检测
export function hasUltrathinkKeyword(text: string): boolean {
  return /\bultrathink\b/i.test(text)
```

```typescript
  if (m.includes('claude')) return false

  model: string,
  userEffort?: EffortValue
): EffortValue | undefined {

  if (userEffort) {

      return 'high'

    return userEffort
  }

  // 3. 默认 undefined (API 默认 high)
  return undefined
}

    case 'medium': return '平衡实现，标准测试'
    case 'high': return '完整实现，全面测试和文档'
```

### 8. Compaction 基础框架

```typescript
// src/compact/index.ts
export interface CompactionResult {
  messages: Message[]
  summary: string
  tokensSaved: number
  compactBoundary: number
}

export async function compactConversation(
  messages: Message[],
  context: CompactContext
): Promise<CompactionResult> {
  // 1. Microcompact 预处理
  const microResult = await microcompactMessages(messages)

  // 2. 分组摘要
  const groups = groupMessagesByTopic(microResult.messages)
  const summaries = await summarizeGroups(groups, context)

  // 3. 构建 compact boundary
  const boundaryIndex = messages.length - summaries.length
  const compactedMessages = [
    createCompactBoundaryMessage(summaries),
    ...summaries,
    ...messages.slice(boundaryIndex)
  ]

  return {
    messages: compactedMessages,
    summary: summaries.map(s => s.content).join('\n'),
    tokensSaved: estimateTokens(messages) - estimateTokens(compactedMessages),
    compactBoundary: boundaryIndex,
  }
}

// Microcompact: 修剪工具结果
function microcompactMessages(messages: Message[]): Message[] {
  return messages.map(msg => {
    if (msg.role === 'tool') {
      const truncated = truncateToolResult(msg.content, MAX_TOOL_RESULT_CHARS)
      return { ...msg, content: truncated }
    }
    return msg
  })
}

function truncateToolResult(content: string, maxChars: number): string {
  if (content.length <= maxChars) return content
  return content.slice(0, maxChars) + '\n...[truncated]'
}
```

```typescript
export interface AutoCompactConfig {
  enabled: boolean
  thresholdPercent: number  // 如 80%
  minMessages: number       // 如 50
}

export function shouldAutoCompact(
  currentTokens: number,
  maxTokens: number,
  messageCount: number,
  config: AutoCompactConfig
): boolean {
  if (!config.enabled) return false
  if (messageCount < config.minMessages) return false

  const utilization = currentTokens / maxTokens
  return utilization >= config.thresholdPercent
}

// 在 query 循环中检查
async function checkAutoCompact(
  messages: Message[],
  model: string,
  context: QueryContext
): Promise<void> {
  const currentTokens = estimateTokens(messages)
  const maxTokens = getModelContextWindow(model)
  const config = getAutoCompactConfig()

  if (shouldAutoCompact(currentTokens, maxTokens, messages.length, config)) {
    console.log('Auto-compacting conversation...')
    await compactConversation(messages, context)
  }
}
```
